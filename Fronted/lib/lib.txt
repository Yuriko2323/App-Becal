/// Muestra URL absoluta, ruta relativa /uploads/... o Base64 legado.
class ProductoImagen extends StatelessWidget {
  final String? origen;
  final BoxFit fit;
  final double? width;
  final double? height;
  final bool permitirZoom;
  const ProductoImagen({super.key, required this.origen, this.fit = BoxFit.contain, this.width, this.height, this.permitirZoom = false});

  ImageProvider? _provider() {
    final raw = origen?.trim();
    if (raw == null || raw.isEmpty) return null;
    if (raw.startsWith('http://') || raw.startsWith('https://')) return NetworkImage(raw);
    if (raw.startsWith('/') || raw.startsWith('uploads/')) {
      final path = raw.startsWith('/') ? raw : '/$raw';
      return NetworkImage('${ApiConfig.baseUrl}$path');
    }
    try {
      var b64 = raw.startsWith('data:') && raw.contains(',') ? raw.substring(raw.indexOf(',') + 1) : raw;
      b64 = b64.replaceAll(RegExp(r'\s+'), '');
      while (b64.length % 4 != 0) { b64 += '='; }
      return MemoryImage(base64Decode(b64));
    } catch (_) { return null; }
  }

  @override
  Widget build(BuildContext context) {
    final provider = _provider();
    if (provider == null) return SizedBox(width: width, height: height, child: const Center(child: Icon(Icons.image_not_supported_outlined, size: 44, color: Colors.grey)));
    final image = Image(image: provider, fit: fit, width: width, height: height, filterQuality: FilterQuality.high,
      errorBuilder: (_, __, ___) => const Center(child: Icon(Icons.broken_image_outlined, size: 44, color: Colors.grey)));
    if (!permitirZoom) return image;
    return GestureDetector(onTap: () => showDialog<void>(context: context, builder: (ctx) => Dialog.fullscreen(
      backgroundColor: Colors.black, child: Stack(children: [
        Center(child: InteractiveViewer(minScale: 0.5, maxScale: 5, child: Image(image: provider, fit: BoxFit.contain))),
        Positioned(right: 12, top: 24, child: IconButton(color: Colors.white, icon: const Icon(Icons.close), onPressed: () => Navigator.pop(ctx))),
      ]))), child: image);
  }
}

class Validators {
  static String? validarCorreo(String? value) {
    if (value == null || value.isEmpty) {
      return 'Por favor, ingresa tu correo electrónico';
    }
    final RegExp emailRegex = RegExp(r'^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$');
    if (!emailRegex.hasMatch(value)) {
      return 'El formato del correo no es válido';
    }
    return null;
  }

  static String? validarContrasena(String? value) {
    if (value == null || value.isEmpty) {
      return 'Por favor, ingresa tu contraseña';
    }
    if (value.length < 6) {
      return 'La contraseña debe tener al menos 6 caracteres';
    }
    return null;
  }
}
String formatoMoneda(num? valor) => '\$${(valor ?? 0).toStringAsFixed(2)}';
String formatoFecha(String? valor) {
  if (valor == null || valor.trim().isEmpty) return 'Sin fecha';
  final fecha = DateTime.tryParse(valor);
  if (fecha == null) return valor;
  return '${fecha.day.toString().padLeft(2, '0')}/${fecha.month.toString().padLeft(2, '0')}/${fecha.year}';
}

class ProductoService {
  Future<List<Producto>> obtenerProductos() async {
    final data = await ApiClient.get('/catalogo/productos');
    if (data is! List) throw Exception('La API devolvió un catálogo con formato inesperado.');
    return data.whereType<Map<String, dynamic>>().map(Producto.fromJson).where((p) => p.activo).toList();
  }
  Future<List<Producto>> obtenerProductosArtesano(int idArtesano) async {
    final data = await ApiClient.get('/artesanos/$idArtesano/productos');
    if (data is! List) throw Exception('La API devolvió una lista de productos inválida.');
    return data.whereType<Map<String, dynamic>>().map(Producto.fromJson).toList();
  }
  Future<List<FotoProducto>> obtenerFotos(int idProducto) async {
    final data = await ApiClient.get('/productos/$idProducto/fotos');
    if (data is! List) throw Exception('La API devolvió una galería inválida.');
    return data.whereType<Map<String, dynamic>>().map(FotoProducto.fromJson).toList()..sort((a,b) => a.orden.compareTo(b.orden));
  }
  Future<dynamic> crearProducto(Map<String, dynamic> body) => ApiClient.post('/productos', body: body);
  Future<dynamic> subirImagen(int idProducto, Uint8List bytes, String nombreArchivo) =>
      ApiClient.uploadFile('/productos/$idProducto/imagen', bytes: bytes, filename: nombreArchivo, fieldName: 'archivo');
  Future<void> actualizarProducto(int id, Map<String, dynamic> body) async => ApiClient.patch('/productos/$id', body: body);
  Future<void> cambiarEstadoProducto(int id, bool activo) async =>
      ApiClient.patch('/productos/$id/estado', body: {'activo': activo});

  Future<void> desactivarProducto(int id) async =>
      cambiarEstadoProducto(id, false);

  Future<void> activarProducto(int id) async =>
      cambiarEstadoProducto(id, true);
}

class PedidoService {
  Future<int> obtenerIdEstadoInicial() async {
    final data = await ApiClient.get('/estados-pedido');
    if (data is! List) throw Exception('No se pudo consultar el catálogo de estados de pedido.');
    final estados = data.whereType<Map<String, dynamic>>().toList();
    if (estados.isEmpty) throw Exception('No hay estados de pedido registrados en la base de datos.');
    final pendiente = estados.where((e) => (e['nombre_estado'] ?? '').toString().toLowerCase().contains('pendiente')).toList();
    if (pendiente.isEmpty) {
      throw Exception('No existe un estado "Pendiente" en el catálogo. El administrador debe registrarlo antes de recibir pedidos.');
    }
    final elegido = pendiente.first;
    final id = elegido['id_estado_pedido'];
    if (id is int) return id;
    final parsed = int.tryParse('$id');
    if (parsed == null) throw Exception('El catálogo de estados no contiene un identificador válido.');
    return parsed;
  }

  Future<Map<String, dynamic>> crearPedido({required int idCliente, required int idProducto, int cantidad = 1, required int idEstadoPedido, String metodoPago = 'Efectivo', String? observaciones}) async {
    final data = await ApiClient.post('/clientes/$idCliente/crear-pedido', body: {
      'id_producto': idProducto, 'cantidad': cantidad, 'id_estado_pedido': idEstadoPedido,
      'metodo_pago': metodoPago, if (observaciones != null && observaciones.trim().isNotEmpty) 'observaciones': observaciones.trim(),
    });
    if (data is! Map<String, dynamic>) throw Exception('La API no devolvió el pedido creado.');
    return data;
  }
  Future<List<Pedido>> obtenerPedidosCliente(int idCliente) async => _lista('/clientes/$idCliente/pedidos-detallados');
  Future<List<Pedido>> obtenerPedidosArtesano(int idArtesano) async => _lista('/artesanos/$idArtesano/pedidos-detallados');
  Future<List<Pedido>> _lista(String path) async {
    final data = await ApiClient.get(path);
    if (data is! List) throw Exception('La API devolvió una lista de pedidos inválida.');
    return data.whereType<Map<String, dynamic>>().map(Pedido.fromJson).where((p) => p.idPedido > 0).toList();
  }
  Future<PedidoDetalle> obtenerDetallePedido(int idCliente, int idPedido) async {
    final data = await ApiClient.get('/clientes/$idCliente/pedidos-detallados/$idPedido');
    if (data is! Map<String, dynamic>) throw Exception('La API devolvió un detalle de pedido inválido.');
    return PedidoDetalle.fromJson(data);
  }
  Future<PedidoDetalle> obtenerDetallePedidoArtesano(int idArtesano, int idPedido) async {
    final data = await ApiClient.get('/artesanos/$idArtesano/pedidos-detallados/$idPedido');
    if (data is! Map<String, dynamic>) throw Exception('La API devolvió un detalle de pedido inválido.');
    return PedidoDetalle.fromJson(data);
  }
  Future<bool> cancelarPedido({required int idCliente, required int idPedido}) async {
    final data = await ApiClient.patch(
      '/clientes/$idCliente/pedidos/$idPedido/cancelar',
      body: <String, dynamic>{},
    );
    if (data is Map && data['cancelado'] == true) return true;
    throw Exception('La API no confirmó la cancelación del pedido.');
  }
}

class ClienteService {
  Future<Cliente> obtenerCliente(int idCliente) async {
    final data = await ApiClient.get('/clientes/$idCliente');
    if (data is! Map<String, dynamic>) throw Exception('Respuesta de perfil inválida.');
    return Cliente.fromJson(data);
  }

  Future<String> obtenerNombreCliente(int idCliente) async {
    try { return (await obtenerCliente(idCliente)).nombreCompleto; }
    catch (_) { return 'Cliente'; }
  }

  Future<bool> actualizarPerfil({
    required int idCliente, required String nombre, required String apellidoPaterno,
    required String apellidoMaterno, required String telefono, required String nacionalidad,
    required String idiomaPreferido, String? fotoPerfilBase64,
  }) async {
    final body = <String, dynamic>{
      'nombre': nombre.trim(), 'apellido_p': apellidoPaterno.trim(),
      'apellido_m': apellidoMaterno.trim(), 'telefono': telefono.trim(),
      'nacionalidad': nacionalidad.trim(), 'idioma_preferido': idiomaPreferido.trim(),
    };
    if (fotoPerfilBase64 != null && fotoPerfilBase64.isNotEmpty) body['foto_perfil'] = fotoPerfilBase64;
    await ApiClient.patch('/clientes/$idCliente', body: body);
    return true;
  }
}

class AuthService {
  Future<Usuario> login(String correo, String contrasena) async {
    final data = await ApiClient.post('/login', body: {
      'correo': correo.trim(),
      'contrasena': contrasena,
    });
    if (data is! Map<String, dynamic>) {
      throw Exception('La API devolvió una respuesta de inicio de sesión inesperada.');
    }
    return Usuario.fromJson(data);
  }

  Future<bool> registrarCliente({
    required String correo,
    required String contrasena,
    required String tipoCliente,
    String nombre = 'Usuario',
    String? apellidoPaterno,
    String? apellidoMaterno,
    String? telefono,
    String? nacionalidad,
    String? idiomaPreferido,
  }) async {
    await ApiClient.post('/registro/cliente', body: {
      'nombre': nombre.trim(),
      'correo': correo.trim(),
      'contrasena': contrasena,
      'tipo_cliente': tipoCliente,
      if (apellidoPaterno != null && apellidoPaterno.trim().isNotEmpty) 'apellido_p': apellidoPaterno.trim(),
      if (apellidoMaterno != null && apellidoMaterno.trim().isNotEmpty) 'apellido_m': apellidoMaterno.trim(),
      if (telefono != null && telefono.trim().isNotEmpty) 'telefono': telefono.trim(),
      if (nacionalidad != null && nacionalidad.trim().isNotEmpty) 'nacionalidad': nacionalidad.trim(),
      if (idiomaPreferido != null && idiomaPreferido.trim().isNotEmpty) 'idioma_preferido': idiomaPreferido.trim(),
    });
    return true;
  }
}

class VendedorDashboardScreen extends StatefulWidget {
  final int idVendedor;
  final String nombre;

  const VendedorDashboardScreen({
    super.key,
    required this.idVendedor,
    required this.nombre,
  });

  @override
  State<VendedorDashboardScreen> createState() =>
      _VendedorDashboardScreenState();
}

class _VendedorDashboardScreenState
    extends State<VendedorDashboardScreen> {
  late Future<List<Map<String, dynamic>>> _futureVentas;

  @override
  void initState() {
    super.initState();
    _futureVentas = _obtenerVentas();
  }

  Future<List<Map<String, dynamic>>> _obtenerVentas() async {
    final respuesta = await ApiClient.get(
      '/ventas?skip=0&limit=100',
    );

    if (respuesta is! List) {
      throw Exception(
        'La API no devolvió una lista válida de ventas.',
      );
    }

    final ventasFiltradas = <Map<String, dynamic>>[];

    for (final elemento in respuesta) {
      if (elemento is! Map) continue;

      final venta = Map<String, dynamic>.from(elemento);
      final idVendedorVenta = venta['id_vendedor'];

      if (idVendedorVenta?.toString() ==
          widget.idVendedor.toString()) {
        ventasFiltradas.add(venta);
      }
    }

    return ventasFiltradas;
  }

  Future<void> _recargarVentas() async {
    final nuevaConsulta = _obtenerVentas();

    setState(() {
      _futureVentas = nuevaConsulta;
    });

    try {
      await nuevaConsulta;
    } catch (_) {
      // FutureBuilder mostrará el error de la consulta.
    }
  }

  void _cerrarSesion() {
    Navigator.pushNamedAndRemoveUntil(
      context,
      '/login',
      (route) => false,
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Vendedor · ${widget.nombre}'),
        actions: [
          IconButton(
            onPressed: _cerrarSesion,
            tooltip: 'Cerrar sesión',
            icon: const Icon(Icons.logout),
          ),
        ],
      ),
      body: FutureBuilder<List<Map<String, dynamic>>>(
        future: _futureVentas,
        builder: (context, snapshot) {
          if (snapshot.connectionState ==
              ConnectionState.waiting) {
            return const LoadingWidget();
          }

          if (snapshot.hasError) {
            return Center(
              child: Padding(
                padding: const EdgeInsets.all(24),
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    const Icon(
                      Icons.cloud_off_outlined,
                      size: 48,
                    ),
                    const SizedBox(height: 12),
                    const Text(
                      'No se pudo consultar el historial de ventas.',
                      textAlign: TextAlign.center,
                    ),
                    const SizedBox(height: 8),
                    Text(
                      '${snapshot.error}',
                      textAlign: TextAlign.center,
                    ),
                    const SizedBox(height: 16),
                    FilledButton.icon(
                      onPressed: _recargarVentas,
                      icon: const Icon(Icons.refresh),
                      label: const Text('Reintentar'),
                    ),
                  ],
                ),
              ),
            );
          }

          final ventas = snapshot.data ??
              <Map<String, dynamic>>[];

          return RefreshIndicator(
            onRefresh: _recargarVentas,
            child: ListView(
              physics: const AlwaysScrollableScrollPhysics(),
              padding: const EdgeInsets.all(16),
              children: [
                Text(
                  'Panel de vendedor',
                  style: Theme.of(context)
                      .textTheme
                      .headlineSmall,
                ),
                const SizedBox(height: 8),
                Text(
                  'Ventas registradas: ${ventas.length}',
                  style: Theme.of(context)
                      .textTheme
                      .titleMedium,
                ),
                const SizedBox(height: 12),

                if (ventas.isEmpty)
                  const Card(
                    child: Padding(
                      padding: EdgeInsets.all(16),
                      child: Text(
                        'No hay ventas asociadas a esta cuenta. '
                        'Si esperabas ver ventas, verifica que la '
                        'respuesta de la API incluya id_vendedor.',
                        textAlign: TextAlign.center,
                      ),
                    ),
                  ),

                ...ventas.map((venta) {
                  final idVenta =
                      venta['id_venta']?.toString() ?? '—';
                  final idPuesto =
                      venta['id_puesto']?.toString() ?? '—';
                  final idCliente =
                      venta['id_cliente']?.toString() ?? '—';
                  final metodoPago =
                      venta['metodo_pago']?.toString() ??
                          'Sin método';

                  return Card(
                    child: ListTile(
                      leading: const Icon(
                        Icons.point_of_sale,
                      ),
                      title: Text('Venta #$idVenta'),
                      subtitle: Text(
                        'Puesto: $idPuesto · Cliente: $idCliente',
                      ),
                      trailing: SizedBox(
                        width: 100,
                        child: Text(
                          metodoPago,
                          textAlign: TextAlign.end,
                          overflow: TextOverflow.ellipsis,
                        ),
                      ),
                    ),
                  );
                }),
              ],
            ),
          );
        },
      ),
    );
  }

class ProductosScreen extends StatefulWidget {
  final int idCliente;
  final VoidCallback? onPedidoCreado;

  const ProductosScreen({
    Key? key,
    required this.idCliente,
    this.onPedidoCreado,
  }) : super(key: key);

  @override
  State<ProductosScreen> createState() => _ProductosScreenState();
}

class _ProductosScreenState extends State<ProductosScreen> {
  final ProductoService _productoService = ProductoService();
  final PedidoService _pedidoService = PedidoService();
  late Future<List<Producto>> _productosFuture;

  @override
  void initState() {
    super.initState();
    _productosFuture = _productoService.obtenerProductos();
  }

  void _mostrarModalDetalle(Producto producto) {
    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (_) => ProductoDetalleScreen(
          producto: producto,
          idCliente: widget.idCliente,
          onComprar: (cantidad) => _procesarCompra(producto, cantidad),
        ),
      ),
    );
  }

  Future<bool> _procesarCompra(Producto producto, int cantidad) async {
    var dialogAbierto = true;
    showDialog<void>(
      context: context,
      barrierDismissible: false,
      builder: (_) => const Center(child: CircularProgressIndicator()),
    );

    try {
      final estadoInicial = await _pedidoService.obtenerIdEstadoInicial();
      final respuesta = await _pedidoService.crearPedido(
        idCliente: widget.idCliente,
        idProducto: producto.idProducto,
        cantidad: cantidad,
        idEstadoPedido: estadoInicial,
        metodoPago: 'Efectivo',
      );

      final rawId = respuesta['id_pedido'];
      final idPedido = rawId is num ? rawId.toInt() : int.tryParse('$rawId');
      if (idPedido == null || idPedido <= 0) {
        throw Exception('La API no confirmó el identificador del pedido. No se puede verificar el registro.');
      }

      if (mounted && dialogAbierto) {
        Navigator.of(context, rootNavigator: true).pop();
        dialogAbierto = false;
      }
      if (!mounted) return false;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Pedido #$idPedido registrado para "${producto.nombre ?? 'Producto'}" (cantidad: $cantidad).'),
          backgroundColor: Colors.green,
        ),
      );
      widget.onPedidoCreado?.call();
      return true;
    } catch (e) {
      if (mounted && dialogAbierto) {
        Navigator.of(context, rootNavigator: true).pop();
        dialogAbierto = false;
      }
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(e.toString().replaceAll('Exception: ', '')),
            backgroundColor: Colors.red,
          ),
        );
      }
      return false;
    }
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<List<Producto>>(
      future: _productosFuture,
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Center(child: CircularProgressIndicator());
        } else if (snapshot.hasError) {
          return Center(
            child: Text(
              'Error: ${snapshot.error.toString().replaceAll('Exception: ', '')}',
              textAlign: TextAlign.center,
              style: const TextStyle(color: Colors.red),
            ),
          );
        } else if (!snapshot.hasData || snapshot.data!.isEmpty) {
          return const Center(
            child: Text('No hay productos disponibles por el momento.'),
          );
        }

        final productos = snapshot.data!;

        return GridView.builder(
          padding: const EdgeInsets.all(16.0),
          gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
            crossAxisCount: 2,
            childAspectRatio: 0.68,
            crossAxisSpacing: 12,
            mainAxisSpacing: 12,
          ),
          itemCount: productos.length,
          itemBuilder: (context, index) {
            return ProductoCard(
              producto: productos[index],
              onComprar: () => _mostrarModalDetalle(productos[index]),
            );
          },
        );
      },
    );
  }
}

class ArtesanoProductosScreen extends StatefulWidget {
  final int idArtesano;

  const ArtesanoProductosScreen({super.key, required this.idArtesano});

  @override
  State<ArtesanoProductosScreen> createState() =>
      _ArtesanoProductosScreenState();

class _ArtesanoProductosScreenState extends State<ArtesanoProductosScreen> {
  final _service = ProductoService();
  late Future<List<Producto>> _future;

  @override
  void initState() {
    super.initState();
    _load();
  }

  void _load() {
    _future = _service.obtenerProductosArtesano(widget.idArtesano);
  }

  int? _toInt(dynamic value) =>
      value is int ? value : int.tryParse('$value');

  List<Map<String, dynamic>> _asMaps(dynamic data, String label) {
    if (data is! List) {
      throw Exception('La API devolvió un catálogo de $label inválido.');
    }
    return data
        .whereType<Map>()
        .map((item) => Map<String, dynamic>.from(item))
        .toList();
  }

  void _message(String value, {bool error = false}) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(value),
        backgroundColor: error ? Colors.red.shade700 : null,
      ),
    );
  }

  Future<void> _form([Producto? product]) async {
    late List<Map<String, dynamic>> styles;
    late List<Map<String, dynamic>> qualities;

    try {
      final results = await Future.wait([
        ApiClient.get('/categorias-estilo?skip=0&limit=200'),
        ApiClient.get('/calidades?skip=0&limit=200'),
      ]);
      styles = _asMaps(results[0], 'estilos');
      qualities = _asMaps(results[1], 'calidades');
    } catch (e) {
      _message('No se pudieron cargar los catálogos: $e', error: true);
      return;
    }

    int? idFrom(Map<String, dynamic> item, String key) =>
        _toInt(item[key]);
    styles = styles.where((item) => idFrom(item, 'id_estilo') != null).toList();
    qualities = qualities
        .where((item) => idFrom(item, 'id_calidad') != null)
        .toList();

    if (styles.isEmpty || qualities.isEmpty) {
      _message(
        'Antes de registrar productos, el administrador debe agregar al menos '
        'un estilo y una calidad en Catálogos del sistema.',
        error: true,
      );
      return;
    }

    final nameController = TextEditingController(text: product?.nombre ?? '');
    final priceController = TextEditingController(
      text: product?.precioBase?.toString() ?? '',
    );
    final descriptionController = TextEditingController(
      text: product?.descripcion ?? '',
    );
    final materialController = TextEditingController(
      text: product?.material ?? '',
    );
    final colorController = TextEditingController(text: product?.color ?? '');
    final sizeController = TextEditingController(text: product?.talla ?? '');

    final styleIds = styles.map((s) => idFrom(s, 'id_estilo')!).toSet();
    final qualityIds = qualities.map((q) => idFrom(q, 'id_calidad')!).toSet();
    int? selectedStyleId =
        styleIds.contains(product?.idEstilo)
            ? product!.idEstilo
            : idFrom(styles.first, 'id_estilo');
    int? selectedQualityId =
        qualityIds.contains(product?.idCalidad)
            ? product!.idCalidad
            : idFrom(qualities.first, 'id_calidad');

    Uint8List? imageBytes;
    String? imageName;
    bool picking = false;

    final saved = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => StatefulBuilder(
        builder: (dialogContext, dialogSetState) => AlertDialog(
          title: Text(
            product == null ? 'Nuevo producto' : 'Editar producto',
          ),
          content: SizedBox(
            width: 460,
            child: SingleChildScrollView(
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  _field(nameController, 'Nombre *'),
                  _field(
                    priceController,
                    'Precio *',
                    keyboard: const TextInputType.numberWithOptions(
                      decimal: true,
                    ),
                  ),
                  _field(descriptionController, 'Descripción', maxLines: 3),
                  _field(materialController, 'Material'),
                  _field(colorController, 'Color'),
                  _field(sizeController, 'Talla'),
                  DropdownButtonFormField<int>(
                    value: selectedStyleId,
                    isExpanded: true,
                    decoration: const InputDecoration(
                      labelText: 'Estilo *',
                      border: OutlineInputBorder(),
                    ),
                    items: styles.map((style) {
                      final id = idFrom(style, 'id_estilo')!;
                      final label =
                          style['nombre_estilo']?.toString().trim().isNotEmpty ==
                                  true
                              ? style['nombre_estilo'].toString()
                              : 'Estilo #$id';
                      return DropdownMenuItem<int>(
                        value: id,
                        child: Text(label, overflow: TextOverflow.ellipsis),
                      );
                    }).toList(),
                    onChanged: (value) =>
                        dialogSetState(() => selectedStyleId = value),
                  ),
                  const SizedBox(height: 12),
                  DropdownButtonFormField<int>(
                    value: selectedQualityId,
                    isExpanded: true,
                    decoration: const InputDecoration(
                      labelText: 'Calidad *',
                      border: OutlineInputBorder(),
                    ),
                    items: qualities.map((quality) {
                      final id = idFrom(quality, 'id_calidad')!;
                      final label =
                          quality['nombre_calidad']?.toString().trim().isNotEmpty ==
                                  true
                              ? quality['nombre_calidad'].toString()
                              : 'Calidad #$id';
                      return DropdownMenuItem<int>(
                        value: id,
                        child: Text(label, overflow: TextOverflow.ellipsis),
                      );
                    }).toList(),
                    onChanged: (value) =>
                        dialogSetState(() => selectedQualityId = value),
                  ),
                  const SizedBox(height: 16),
                  Align(
                    alignment: Alignment.centerLeft,
                    child: Text(
                      'Fotografía del producto',
                      style: Theme.of(dialogContext).textTheme.titleSmall,
                    ),
                  ),
                  const SizedBox(height: 8),
                  if (imageBytes != null)
                    ClipRRect(
                      borderRadius: BorderRadius.circular(8),
                      child: Image.memory(
                        imageBytes!,
                        height: 140,
                        width: double.infinity,
                        fit: BoxFit.contain,
                      ),
                    )
                  else if ((product?.imagenUrl ?? '').trim().isNotEmpty)
                    SizedBox(
                      height: 110,
                      width: double.infinity,
                      child: ProductoImagen(
                        origen: product!.imagenUrl,
                        fit: BoxFit.contain,
                      ),
                    ),
                  if (imageName != null)
                    Padding(
                      padding: const EdgeInsets.only(top: 6),
                      child: Text(
                        imageName!,
                        maxLines: 2,
                        overflow: TextOverflow.ellipsis,
                      ),
                    ),
                  const SizedBox(height: 8),
                  OutlinedButton.icon(
                    onPressed: picking
                        ? null
                        : () async {
                            dialogSetState(() => picking = true);
                            try {
                              final result =
                                  await FilePicker.platform.pickFiles(
                                type: FileType.custom,
                                allowedExtensions: const [
                                  'jpg',
                                  'jpeg',
                                  'png',
                                  'webp',
                                ],
                                allowMultiple: false,
                                withData: true,
                              );
                              if (result != null && result.files.isNotEmpty) {
                                final file = result.files.single;
                                if (file.size > 5 * 1024 * 1024) {
                                  _message(
                                    'La imagen supera 5 MB. Selecciona una '
                                    'imagen más pequeña.',
                                    error: true,
                                  );
                                } else if (file.bytes == null ||
                                    file.bytes!.isEmpty) {
                                  _message(
                                    'No se pudo leer el archivo seleccionado.',
                                    error: true,
                                  );
                                } else {
                                  dialogSetState(() {
                                    imageBytes = file.bytes;
                                    imageName = file.name;
                                  });
                                }
                              }
                            } catch (e) {
                              _message(
                                'No se pudo abrir el selector de archivos: $e',
                                error: true,
                              );
                            } finally {
                              dialogSetState(() => picking = false);
                            }
                          },
                    icon: picking
                        ? const SizedBox(
                            width: 16,
                            height: 16,
                            child: CircularProgressIndicator(strokeWidth: 2),
                          )
                        : const Icon(Icons.upload_file),
                    label: Text(
                      picking
                          ? 'Abriendo archivos...'
                          : 'Elegir imagen desde archivos',
                    ),
                  ),
                  const Align(
                    alignment: Alignment.centerLeft,
                    child: Text(
                      'JPG, PNG o WEBP. Máximo 5 MB. La foto se guarda en el servidor.',
                      style: TextStyle(fontSize: 12, color: Colors.grey),
                    ),
                  ),
                ],
              ),
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext, false),
              child: const Text('Cancelar'),
            ),
            FilledButton(
              onPressed: () => Navigator.pop(dialogContext, true),
              child: const Text('Guardar'),
            ),
          ],
        ),
      ),
    );

    if (saved != true) {
      _disposeControllers([
        nameController,
        priceController,
        descriptionController,
        materialController,
        colorController,
        sizeController,
      ]);
      return;
    }

    final nameValue = nameController.text.trim();
    final priceValue = double.tryParse(
      priceController.text.replaceAll(',', '.').trim(),
    );
    if (nameValue.isEmpty ||
        priceValue == null ||
        priceValue <= 0 ||
        selectedStyleId == null ||
        selectedQualityId == null) {
      _message(
        'Revisa los campos obligatorios: nombre, precio mayor que cero, estilo y calidad.',
        error: true,
      );
      _disposeControllers([
        nameController,
        priceController,
        descriptionController,
        materialController,
        colorController,
        sizeController,
      ]);
      return;
    }

    final body = <String, dynamic>{
      'nombre': nameValue,
      'precio_base': priceValue,
      'descripcion': _nullableText(descriptionController.text),
      'material': _nullableText(materialController.text),
      'color': _nullableText(colorController.text),
      'talla': _nullableText(sizeController.text),
      'id_estilo': selectedStyleId,
      'id_calidad': selectedQualityId,
      'imagen_url': product?.imagenUrl,
      if (product == null) 'id_artesano': widget.idArtesano,
      // No enviamos id_producto: SQL Server lo genera como IDENTITY.
    };
    _disposeControllers([
      nameController,
      priceController,
      descriptionController,
      materialController,
      colorController,
      sizeController,
    ]);

    var productSaved = false;
    try {
      int productId;
      if (product == null) {
        final created = await _service.crearProducto(body);
        productSaved = true;
        productId = created is Map ? (_toInt(created['id_producto']) ?? 0) : 0;
      } else {
        await _service.actualizarProducto(product.idProducto, body);
        productSaved = true;
        productId = product.idProducto;
      }

      if (imageBytes != null) {
        if (productId <= 0) {
          throw Exception(
            'El producto se guardó, pero la API no devolvió id_producto; '
            'no se pudo asociar la imagen.',
          );
        }
        await _service.subirImagen(
          productId,
          imageBytes!,
          imageName ?? 'producto.jpg',
        );
      }

      _message(
        imageBytes == null
            ? 'Producto guardado.'
            : 'Producto e imagen guardados.',
      );
      if (mounted) setState(_load);
    } catch (e) {
      _message(
        productSaved
            ? 'El producto se guardó, pero hubo un problema con la imagen o la respuesta de la API: $e'
            : 'No se pudo guardar el producto: $e',
        error: true,
      );
      if (productSaved && mounted) setState(_load);
    }
  }

  String? _nullableText(String value) {
    final trimmed = value.trim();
    return trimmed.isEmpty ? null : trimmed;
  }

  void _disposeControllers(List<TextEditingController> controllers) {
    for (final controller in controllers) {
      controller.dispose();
    }
  }

  Future<void> _changeProductState(Producto product, bool active) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: Text(active ? 'Activar producto' : 'Desactivar producto'),
        content: Text(
          active
              ? '¿Volver a publicar “${product.nombre}” en el catálogo?'
              : '¿Ocultar “${product.nombre}” del catálogo? No se eliminará físicamente.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('Cancelar'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            child: Text(active ? 'Activar' : 'Desactivar'),
          ),
        ],
      ),
    );
    if (confirmed != true) return;

    try {
      await _service.cambiarEstadoProducto(product.idProducto, active);
      _message(active ? 'Producto activado.' : 'Producto desactivado.');
      if (mounted) setState(_load);
    } catch (e) {
      _message(
        'No se pudo ${active ? 'activar' : 'desactivar'} el producto: $e. '
        'Verifica que se haya agregado la ruta de estado a la API.',
        error: true,
      );
    }
  }

            if (snapshot.connectionState == ConnectionState.waiting) {

                        'Error al cargar productos: ${snapshot.error}',

                      ),
                      const SizedBox(height: 12),
                      FilledButton(
                        onPressed: () => setState(_load),

                ),

            final products = snapshot.data ?? [];
            return RefreshIndicator(

                padding: const EdgeInsets.all(12),
                children: [

                    label: const Text('Agregar producto'),

                  const SizedBox(height: 8),
                  if (products.isEmpty)
                    const Padding(
                      padding: EdgeInsets.all(24),
                      child: Text(
                        'No tienes productos registrados.',
                        textAlign: TextAlign.center,
                      ),
                    ),
                  ...products.map((product) => Card(

                          leading: SizedBox(
                            width: 60,

                              fit: BoxFit.contain,

                          ),
                          title: Text(product.nombre ?? 'Sin nombre'),
                          subtitle: Text(
                            '${formatoMoneda(product.precioBase)} · '
                            '${product.activo ? 'Activo' : 'Inactivo'}',
                          ),
                          trailing: Wrap(
                            spacing: 0,
                            children: [
                              IconButton(
                                tooltip: 'Editar',
                                onPressed: () => _form(product),
                                icon: const Icon(Icons.edit_outlined),

                              IconButton(
                                tooltip: product.activo
                                    ? 'Desactivar producto'
                                    : 'Volver a activar producto',
                                onPressed: () => _changeProductState(
                                  product,
                                  !product.activo,
                                ),
                                icon: Icon(
                                  product.activo
                                      ? Icons.visibility_off_outlined
                                      : Icons.visibility_outlined,
                                ),
                                color: product.activo
                                    ? Colors.red.shade700
                                    : Colors.green.shade700,
                              ),
                            ],
                          ),
                        ),
                      )),
                ],
              ),
            );
          },
        ),
      );

  static Widget _field(

class _PerfilScreenState extends State<PerfilScreen> {
  final ClienteService _clienteService = ClienteService();
  late Future<Cliente> _clienteFuture;

  @override
  void initState() {
    super.initState();
    _cargarDatos();
  }

  void _cargarDatos() {
    setState(() {
      _clienteFuture = _clienteService.obtenerCliente(widget.idCliente);
    });
  }

  Widget _buildAvatar(Cliente cliente) {
    ImageProvider? imageProvider;

    if (cliente.fotoPerfil != null && cliente.fotoPerfil!.trim().isNotEmpty) {
      final foto = cliente.fotoPerfil!.trim();

      if (foto.startsWith('http')) {
        imageProvider = NetworkImage(foto);
      } else {
        try {
          String cleanBase64 = foto.contains(',') ? foto.split(',').last : foto;
          cleanBase64 = cleanBase64.replaceAll('\n', '').replaceAll('\r', '').replaceAll(' ', '');

          while (cleanBase64.length % 4 != 0) {
            cleanBase64 += '=';
          }

          imageProvider = MemoryImage(base64Decode(cleanBase64));
        } catch (e) {
          debugPrint('Error decodificando foto de perfil: $e');
          imageProvider = null;
        }
      }
    }

    return CircleAvatar(
      radius: 50,
      backgroundColor: Colors.deepOrange.shade100,
      backgroundImage: imageProvider,
      child: imageProvider == null
          ? const Icon(Icons.person, size: 55, color: Colors.deepOrange)
          : null,
    );
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<Cliente>(
      future: _clienteFuture,

          child: Column(
            children: [
              const SizedBox(height: 10),
              _buildAvatar(cliente),
              const SizedBox(height: 16),
              Text(

                textAlign: TextAlign.center,
                style: const TextStyle(fontSize: 22, fontWeight: FontWeight.bold),
              ),
              const SizedBox(height: 4),
              Chip(
                label: Text(
                  'Tipo: ${cliente.tipoCliente ?? 'Sin especificar'}',
                  style: const TextStyle(fontWeight: FontWeight.w600),
                ),
                backgroundColor: Colors.deepOrange.shade50,
                side: BorderSide(color: Colors.deepOrange.shade200),
              ),
              const SizedBox(height: 16),
              const Divider(),
              const SizedBox(height: 12),

              const Align(
                alignment: Alignment.centerLeft,
                child: Text(
                  'Mi información',
                  style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
                ),
              ),
              const SizedBox(height: 12),

              ListTile(
                leading: const Icon(Icons.phone),

                title: const Text('Idioma preferido'),
                subtitle: Text(
                  cliente.idiomaPreferido != null && cliente.idiomaPreferido!.isNotEmpty

                      : 'No especificado',

              const SizedBox(height: 20),
              const Divider(),
              const SizedBox(height: 12),

                title: const Text(
                  'Editar perfil',

                  Navigator.pushNamedAndRemoveUntil(context, '/login', (route) => false);

class _EditarPerfilScreenState extends State<EditarPerfilScreen> {
  final _formKey = GlobalKey<FormState>();
  final ClienteService _clienteService = ClienteService();

  late TextEditingController _nombreController;
  late TextEditingController _apellidoPController;
  late TextEditingController _apellidoMController;
  late TextEditingController _telefonoController;
  late TextEditingController _nacionalidadController;
  late TextEditingController _idiomaController;

  String? _fotoBase64;
  bool _isLoading = false;

  @override
  void initState() {
    super.initState();
    _nombreController = TextEditingController(text: widget.cliente.nombre ?? '');
    _apellidoPController = TextEditingController(text: widget.cliente.apellidoPaterno ?? '');
    _apellidoMController = TextEditingController(text: widget.cliente.apellidoMaterno ?? '');
    _telefonoController = TextEditingController(text: widget.cliente.telefono ?? '');
    _nacionalidadController = TextEditingController(text: widget.cliente.nacionalidad ?? 'Mexicana');
    _idiomaController = TextEditingController(text: widget.cliente.idiomaPreferido ?? 'Español');
    _fotoBase64 = widget.cliente.fotoPerfil;
  }

  Future<void> _seleccionarFoto() async {
    try {
      final result = await FilePicker.platform.pickFiles(
        type: FileType.image,
        allowMultiple: false,
        withData: true,
      );

      if (result != null && result.files.isNotEmpty) {
        final Uint8List? fileBytes = result.files.first.bytes;

        if (fileBytes != null) {
          final String base64String = base64Encode(fileBytes);

          setState(() {
            _fotoBase64 = 'data:image/png;base64,$base64String';
          });
        }
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Error al seleccionar la imagen: $e')),
        );
      }
    }
  }

  void _guardarCambios() async {
    if (!_formKey.currentState!.validate()) return;

    setState(() => _isLoading = true);

    try {
      final exito = await _clienteService.actualizarPerfil(
        idCliente: widget.cliente.idCliente,
        nombre: _nombreController.text,
        apellidoPaterno: _apellidoPController.text,
        apellidoMaterno: _apellidoMController.text,
        telefono: _telefonoController.text,
        nacionalidad: _nacionalidadController.text,
        idiomaPreferido: _idiomaController.text,
        fotoPerfilBase64: _fotoBase64,
      );

      if (exito && mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('Perfil actualizado correctamente.'),
            backgroundColor: Colors.green,
          ),
        );
        Navigator.pop(context, true);
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(e.toString().replaceAll('Exception: ', '')),
            backgroundColor: Colors.red,
          ),
        );
      }
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Widget _buildAvatar() {
    ImageProvider? imageProvider;
    if (_fotoBase64 != null && _fotoBase64!.trim().isNotEmpty) {
      final foto = _fotoBase64!.trim();
      if (foto.startsWith('http')) {
        imageProvider = NetworkImage(foto);
      } else {
        try {
          String cleanBase64 = foto.contains(',') ? foto.split(',').last : foto;
          cleanBase64 = cleanBase64.replaceAll('\n', '').replaceAll('\r', '').replaceAll(' ', '');
          while (cleanBase64.length % 4 != 0) {
            cleanBase64 += '=';
          }
          imageProvider = MemoryImage(base64Decode(cleanBase64));
        } catch (_) {
          imageProvider = null;
        }
      }
    }

    return Center(
      child: Stack(
        children: [
          CircleAvatar(
            radius: 50,
            backgroundColor: Colors.deepOrange.shade100,
            backgroundImage: imageProvider,
            child: imageProvider == null
                ? const Icon(Icons.person, size: 60, color: Colors.deepOrange)
                : null,
          ),
          Positioned(
            bottom: 0,
            right: 0,
            child: CircleAvatar(
              radius: 18,
              backgroundColor: Theme.of(context).primaryColor,
              child: IconButton(
                icon: const Icon(Icons.camera_alt, size: 18, color: Colors.white),
                onPressed: _seleccionarFoto,
              ),
            ),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Editar Perfil')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20.0),
        child: Form(
          key: _formKey,
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              _buildAvatar(),
              const SizedBox(height: 24),
              const Text(
                'Información Personal',
                style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold),
              ),
              const SizedBox(height: 16),
              TextFormField(
                controller: _nombreController,
                decoration: const InputDecoration(
                  labelText: 'Nombre(s)',
                  border: OutlineInputBorder(),
                  prefixIcon: Icon(Icons.person),
                ),
                validator: (v) => v == null || v.trim().isEmpty ? 'Ingresa tu nombre' : null,
              ),
              const SizedBox(height: 16),
              Row(
                children: [
                  Expanded(
                    child: TextFormField(
                      controller: _apellidoPController,
                      decoration: const InputDecoration(
                        labelText: 'Apellido Paterno',
                        border: OutlineInputBorder(),
                      ),
                      validator: (v) => v == null || v.trim().isEmpty ? 'Requerido' : null,
                    ),
                  ),
                  const SizedBox(width: 12),
                  Expanded(
                    child: TextFormField(
                      controller: _apellidoMController,
                      decoration: const InputDecoration(
                        labelText: 'Apellido Materno',
                        border: OutlineInputBorder(),
                      ),
                    ),
                  ),
                ],
              ),
              const SizedBox(height: 16),
              TextFormField(
                controller: _telefonoController,
                keyboardType: TextInputType.phone,
                decoration: const InputDecoration(
                  labelText: 'Teléfono de contacto',
                  border: OutlineInputBorder(),

              const SizedBox(height: 24),

              const SizedBox(height: 16),
              TextFormField(
                controller: _nacionalidadController,
                decoration: const InputDecoration(
                  labelText: 'Nacionalidad',
                  border: OutlineInputBorder(),
                  prefixIcon: Icon(Icons.flag),
                ),

              const SizedBox(height: 16),
              TextFormField(

    setState(() {

    showDialog(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Cancelar Pedido'),
        content: Text('¿Estás seguro de que deseas cancelar el pedido #$idPedido?'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext),
            child: const Text('Volver'),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(backgroundColor: Colors.red, foregroundColor: Colors.white),
            onPressed: () async {
              Navigator.pop(dialogContext);
              _procesarCancelacion(idPedido);
            },

          ),

class _LoginScreenState extends State<LoginScreen> {
  final _correoController = TextEditingController();
  final _contrasenaController = TextEditingController();
  final AuthService _authService = AuthService();
  bool _isLoading = false;

  void _iniciarSesion() async {
    if (_correoController.text.trim().isEmpty || _contrasenaController.text.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Por favor llena todos los campos')),
      );
      return;
    }

    setState(() => _isLoading = true);

    try {
      final usuario = await _authService.login(
        _correoController.text.trim(),
        _contrasenaController.text,
      );

      if (!mounted) return;
      switch (usuario.tipoUsuario) {
        case 'Admin':
          Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => const AdminDashboardScreen()));
          break;
        case 'Cliente':
          final idCliente = usuario.idCliente;
          if (idCliente == null) {
            throw Exception('La API autenticó la cuenta, pero no devolvió id_cliente. Revisa el perfil en la base de datos.');
          }
          Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => HomeScreen(idCliente: idCliente)));
          break;
        case 'Artesano':
          final id = usuario.idArtesano;
          if (id == null) throw Exception('La API no devolvió id_artesano para esta cuenta.');
          Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => ArtesanoDashboardScreen(idArtesano: id, nombre: usuario.nombre)));
          break;
        case 'Vendedor':
          final id = usuario.idVendedor;
          if (id == null) throw Exception('La API no devolvió id_vendedor para esta cuenta.');
          Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => VendedorDashboardScreen(idVendedor: id, nombre: usuario.nombre)));
          break;
        default:
          throw Exception('Tipo de usuario no reconocido: ${usuario.tipoUsuario}');
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(e.toString().replaceAll('Exception: ', '')),
            backgroundColor: Colors.red,
          ),
        );
      }
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Padding(
        padding: const EdgeInsets.all(24.0),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            const Text(
              'Parador Artesanal',
              style: TextStyle(fontSize: 28, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 40),
            TextField(
              controller: _correoController,
              keyboardType: TextInputType.emailAddress,
              decoration: const InputDecoration(
                labelText: 'Correo electrónico',
                border: OutlineInputBorder(),
                prefixIcon: Icon(Icons.email),
              ),
            ),
            const SizedBox(height: 20),
            TextField(
              controller: _contrasenaController,
              obscureText: true,
              decoration: const InputDecoration(
                labelText: 'Contraseña',
                border: OutlineInputBorder(),
                prefixIcon: Icon(Icons.lock),
              ),
            ),
            const SizedBox(height: 30),
            _isLoading
                ? const CircularProgressIndicator()
                : ElevatedButton(
                    onPressed: _iniciarSesion,
                    style: ElevatedButton.styleFrom(
                      minimumSize: const Size(double.infinity, 50),
                    ),
                    child: const Text('Iniciar Sesión'),
                  ),
            TextButton(
              onPressed: () => Navigator.pushNamed(context, '/registro'),
              child: const Text('Registrarse'),
            )
          ],
        ),
      ),
    );
  }
}

  String _subtipoCliente = 'Local'; // 'Local', 'Mayorista', 'Turista'
  bool _isLoading = false;

  void _registrar() async {
    if (!_formKey.currentState!.validate()) return;

    if (_contrasenaController.text != _confirmarContrasenaController.text) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Las contraseñas no coinciden')),
      );
      return;
    }

    // Nota: De acuerdo a las reglas de negocio de tu sistema, las cuentas de
    // Artesano y Vendedor se asignan desde el panel de Administración.
    if (_tipoUsuario != 'Cliente') {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            'Las cuentas de $_tipoUsuario se gestionan a través de la administración del Parador.',
          ),
        ),
      );
      return;
    }

    try {
      final exito = await _authService.registrarCliente(
        correo: _correoController.text.trim(),
        contrasena: _contrasenaController.text,
        tipoCliente: _subtipoCliente,

          const SnackBar(
            content: Text('¡Cuenta creada con éxito! Por favor inicia sesión.'),

            content: Text(e.toString().replaceAll('Exception: ', '')),
            backgroundColor: Colors.red,

    }

  late final TextEditingController _apellidoPaterno = TextEditingController(text: _s('apellido_paterno'));
  late final TextEditingController _apellidoMaterno = TextEditingController(text: _s('apellido_materno'));
  late final TextEditingController _correo = TextEditingController(text: _s('correo'));
  late final TextEditingController _telefono = TextEditingController(text: _s('telefono'));
  late final TextEditingController _biografia = TextEditingController(text: _s('biografia'));
  late final TextEditingController _experiencia = TextEditingController(text: _s('anios_experiencia', '0'));
  Uint8List? _fotoBytes;
  String? _fotoNombre;
  bool _saving = false;

  ImageProvider? _fotoActual() {
    final raw = _s('foto_perfil').trim();
    if (raw.isEmpty) return null;
    if (raw.startsWith('http://') || raw.startsWith('https://')) return NetworkImage(raw);
    if (raw.startsWith('/')) return NetworkImage('${ApiConfig.baseUrl}$raw');
    try {
      final encoded = raw.startsWith('data:') && raw.contains(',') ? raw.substring(raw.indexOf(',') + 1) : raw;
      return MemoryImage(base64Decode(encoded.replaceAll(RegExp(r'\s+'), '')));
    } catch (_) { return null; }
  }

  String _s(String key, [String fallback = '']) => widget.perfil[key]?.toString() ?? fallback;
  String? _optional(String value) => value.trim().isEmpty ? null : value.trim();

  Future<void> _pickPhoto() async {
    try {
      final result = await FilePicker.platform.pickFiles(type: FileType.custom, allowedExtensions: const ['jpg', 'jpeg', 'png', 'webp'], allowMultiple: false, withData: true);
      if (result == null || result.files.isEmpty) return;
      final file = result.files.single;
      if (file.size > 5 * 1024 * 1024) {
        _message('La foto no puede superar 5 MB.', error: true);
        return;
      }
      if (file.bytes == null || file.bytes!.isEmpty) {
        _message('No se pudo leer la imagen seleccionada.', error: true);
        return;
      }
      setState(() { _fotoBytes = file.bytes; _fotoNombre = file.name; });
    } catch (e) { _message('No se pudo abrir el selector de archivos: $e', error: true); }
  }

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    final correo = _correo.text.trim();
    if (!correo.contains('@') || !correo.contains('.')) { _message('Escribe un correo válido.', error: true); return; }
    setState(() => _saving = true);
    bool datosGuardados = false;
    try {
      await ApiClient.patch('/artesanos/${widget.idArtesano}/perfil', body: {
        'nombre': _nombre.text.trim(),
        'apellido_paterno': _optional(_apellidoPaterno.text),
        'apellido_materno': _optional(_apellidoMaterno.text),
        'correo': correo,
        'telefono': _optional(_telefono.text),
        'biografia': _optional(_biografia.text),
        'anios_experiencia': int.parse(_experiencia.text.trim()),
      });
      datosGuardados = true;
      if (_fotoBytes != null) {
        await ApiClient.uploadFile('/artesanos/${widget.idArtesano}/foto-perfil', bytes: _fotoBytes!, filename: _fotoNombre ?? 'perfil.jpg', fieldName: 'archivo');
      }
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Perfil actualizado correctamente.'), backgroundColor: Colors.green));
      Navigator.pop(context, true);
    } catch (e) {
      if (!mounted) return;
      final prefijo = datosGuardados ? 'Tus datos se guardaron, pero la foto no pudo actualizarse: ' : 'No se pudo guardar el perfil: ';
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('$prefijo$e'), backgroundColor: Colors.red.shade700));
    } finally { if (mounted) setState(() => _saving = false); }
  }

  void _message(String value, {bool error = false}) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(value), backgroundColor: error ? Colors.red.shade700 : null));
  }

  @override
  void dispose() {
    _nombre.dispose(); _apellidoPaterno.dispose(); _apellidoMaterno.dispose(); _correo.dispose(); _telefono.dispose(); _biografia.dispose(); _experiencia.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: const Text('Editar mi perfil')),
    body: Form(
      key: _formKey,
      child: ListView(padding: const EdgeInsets.all(16), children: [
        Center(child: Column(children: [
          CircleAvatar(radius: 52, backgroundColor: Colors.deepOrange.shade100, backgroundImage: _fotoBytes != null ? MemoryImage(_fotoBytes!) : _fotoActual(), child: (_fotoBytes == null && _fotoActual() == null) ? const Icon(Icons.person, size: 58, color: Colors.deepOrange) : null),
          const SizedBox(height: 8),
          OutlinedButton.icon(onPressed: _saving ? null : _pickPhoto, icon: const Icon(Icons.add_a_photo_outlined), label: const Text('Elegir foto de perfil')),
          if (_fotoNombre != null) Text(_fotoNombre!, maxLines: 1, overflow: TextOverflow.ellipsis),
          const Text('JPG, PNG o WEBP; máximo 5 MB.', style: TextStyle(fontSize: 12, color: Colors.grey)),
        ])),
        const SizedBox(height: 20),
        _field(_nombre, 'Nombre *', required: true, icon: Icons.person_outline),
        _field(_apellidoPaterno, 'Apellido paterno', icon: Icons.person_outline),
        _field(_apellidoMaterno, 'Apellido materno', icon: Icons.person_outline),
        _field(_correo, 'Correo de acceso *', required: true, keyboard: TextInputType.emailAddress, icon: Icons.email_outlined),
        _field(_telefono, 'Teléfono', keyboard: TextInputType.phone, icon: Icons.phone_outlined),
        _field(_biografia, 'Biografía del taller', maxLines: 4, icon: Icons.description_outlined),
        _field(_experiencia, 'Años de experiencia *', required: true, keyboard: TextInputType.number, icon: Icons.work_outline, validator: (v) { final n = int.tryParse((v ?? '').trim()); if (n == null || n < 0) return 'Escribe un número mayor o igual que cero.'; return null; }),
        const SizedBox(height: 4),
        const Text('La cueva, el puesto y los identificadores no se modifican desde el perfil.', style: TextStyle(color: Colors.grey, fontSize: 12)),
        const SizedBox(height: 16),
        FilledButton.icon(onPressed: _saving ? null : _save, icon: _saving ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2)) : const Icon(Icons.save_outlined), label: Text(_saving ? 'Guardando...' : 'Guardar cambios'), style: FilledButton.styleFrom(minimumSize: const Size.fromHeight(50))),
      ]),
    ),
  );

  Widget _field(TextEditingController controller, String label, {bool required = false, TextInputType? keyboard, int maxLines = 1, IconData? icon, String? Function(String?)? validator}) => Padding(
    padding: const EdgeInsets.only(bottom: 12),
    child: TextFormField(controller: controller, keyboardType: keyboard, maxLines: maxLines, decoration: InputDecoration(labelText: label, prefixIcon: icon == null ? null : Icon(icon), border: const OutlineInputBorder()), validator: validator ?? (required ? (v) => v == null || v.trim().isEmpty ? 'Este campo es obligatorio.' : null : null)),
  );
}

class _ArtesanoPerfilScreenState extends State<ArtesanoPerfilScreen> {
  late Future<Map<String, dynamic>> _future = _cargarPerfil();

  Future<Map<String, dynamic>> _cargarPerfil() async {
    final result = await ApiClient.get('/artesanos/${widget.idArtesano}/perfil');
    if (result is! Map) throw Exception('La API devolvió un perfil inválido.');
    return Map<String, dynamic>.from(result);
  }

  Future<void> _recargar() async {
    final next = _cargarPerfil();
    setState(() => _future = next);
    try { await next; } catch (_) {}
  }

  Future<void> _editar(Map<String, dynamic> perfil) async {
    final result = await Navigator.push<bool>(context, MaterialPageRoute(
      builder: (_) => ArtesanoEditarPerfilScreen(idArtesano: widget.idArtesano, perfil: perfil),
    ));
    if (result == true && mounted) await _recargar();
  }

  ImageProvider? _imageProvider(String? value) {
    final raw = value?.trim() ?? '';
    if (raw.isEmpty) return null;
    if (raw.startsWith('http://') || raw.startsWith('https://')) return NetworkImage(raw);
    if (raw.startsWith('/')) return NetworkImage('${ApiConfig.baseUrl}$raw');
    try {
      final encoded = raw.startsWith('data:') && raw.contains(',') ? raw.substring(raw.indexOf(',') + 1) : raw;
      return MemoryImage(base64Decode(encoded.replaceAll(RegExp(r'\s+'), '')));
    } catch (_) { return null; }
  }

  String _value(dynamic value, [String fallback = 'No especificado']) {
    final text = value?.toString().trim() ?? '';
    return text.isEmpty ? fallback : text;
  }

  @override
  Widget build(BuildContext context) => FutureBuilder<Map<String, dynamic>>(
    future: _future,
    builder: (context, snapshot) {
      if (snapshot.connectionState == ConnectionState.waiting) return const LoadingWidget();
      if (snapshot.hasError) return Center(child: Padding(padding: const EdgeInsets.all(24), child: Column(mainAxisSize: MainAxisSize.min, children: [
        const Icon(Icons.cloud_off_outlined, size: 44),
        const SizedBox(height: 12),
        Text('No se pudo cargar tu perfil.\n${snapshot.error}', textAlign: TextAlign.center),
        const SizedBox(height: 12),
        FilledButton.icon(onPressed: _recargar, icon: const Icon(Icons.refresh), label: const Text('Reintentar')),
      ])));
      final p = snapshot.data!;
      final nombre = '${_value(p['nombre'], '')} ${_value(p['apellido_paterno'], '')} ${_value(p['apellido_materno'], '')}'.trim();
      final avatar = _imageProvider(p['foto_perfil']?.toString());
      final experiencia = p['anios_experiencia']?.toString() ?? '0';
      return RefreshIndicator(
        onRefresh: _recargar,
        child: ListView(physics: const AlwaysScrollableScrollPhysics(), padding: const EdgeInsets.all(16), children: [
          Text('Mi perfil', style: Theme.of(context).textTheme.headlineSmall?.copyWith(fontWeight: FontWeight.bold)),
          const SizedBox(height: 14),
          Card(child: Padding(padding: const EdgeInsets.all(20), child: Column(children: [
            CircleAvatar(radius: 48, backgroundColor: Colors.deepOrange.shade100, backgroundImage: avatar, child: avatar == null ? const Icon(Icons.person, size: 56, color: Colors.deepOrange) : null),
            const SizedBox(height: 12),
            Text(nombre.isEmpty ? 'Artesano de Bécal' : nombre, textAlign: TextAlign.center, style: Theme.of(context).textTheme.titleLarge?.copyWith(fontWeight: FontWeight.bold)),
            const SizedBox(height: 4),
            Text(_value(p['correo']), style: TextStyle(color: Colors.grey.shade700)),
            const SizedBox(height: 12),
            Wrap(spacing: 8, runSpacing: 8, alignment: WrapAlignment.center, children: [
              Chip(avatar: const Icon(Icons.work_outline, size: 18), label: Text('$experiencia años de experiencia')),
              Chip(avatar: const Icon(Icons.terrain, size: 18), label: Text(_value(p['cueva'], 'Sin cueva'))),
              Chip(avatar: const Icon(Icons.storefront_outlined, size: 18), label: Text(_value(p['puesto'], 'Sin puesto'))),
            ]),
            const SizedBox(height: 8),
            SizedBox(width: double.infinity, child: FilledButton.icon(onPressed: () => _editar(p), icon: const Icon(Icons.edit_outlined), label: const Text('Editar mi perfil'))),
          ]))),
          const SizedBox(height: 12),
          Card(child: Padding(padding: const EdgeInsets.all(16), child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
            Text('Resumen de mi taller', style: Theme.of(context).textTheme.titleMedium?.copyWith(fontWeight: FontWeight.bold)),
            const SizedBox(height: 10),
            _detail('Teléfono', _value(p['telefono'])),
            _detail('Años de experiencia', experiencia),
            _detail('Fecha de registro', _value(p['fecha_registro'])),
            _detail('Biografía', _value(p['biografia'])),
          ]))),
          const SizedBox(height: 8),
          Card(child: Padding(padding: const EdgeInsets.all(16), child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
            Text('Datos de asignación', style: Theme.of(context).textTheme.titleMedium?.copyWith(fontWeight: FontWeight.bold)),
            const SizedBox(height: 10),
            _detail('ID de artesano', '${p['id_artesano'] ?? widget.idArtesano}'),
            _detail('Cueva', _value(p['cueva'])),
            _detail('Dirección de cueva', _value(p['direccion_cueva'])),
            _detail('Puesto', _value(p['puesto'])),
            _detail('Número de puesto', _value(p['numero_puesto'])),
            const SizedBox(height: 6),
            const Text('Los identificadores y la asignación de cueva y puesto se administran desde el panel correspondiente.', style: TextStyle(color: Colors.grey, fontSize: 12)),
          ]))),
        ]),
      );
    },
  );

  Widget _detail(String label, String value) => Padding(padding: const EdgeInsets.symmetric(vertical: 5), child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [SizedBox(width: 150, child: Text(label, style: TextStyle(color: Colors.grey.shade700)),), Expanded(child: Text(value))]));

import 'package:flutter/material.dart';

class AdminVentasScreen extends StatefulWidget {
  const AdminVentasScreen({super.key});
  @override
  State<AdminVentasScreen> createState() => _AdminVentasScreenState();
}

class _AdminVentasScreenState extends State<AdminVentasScreen> {
  bool _loading = true;
  String? _error;
  List<Map<String, dynamic>> _ventas = [];
  double _totalGeneral = 0;
  int _cantidadVentas = 0;
  int _totalPiezas = 0;

  @override
  void initState() { super.initState(); _load(); }

  int? _int(dynamic v) => v is int ? v : (v is num ? v.toInt() : int.tryParse('$v'));
  double _num(dynamic v) => v is num ? v.toDouble() : double.tryParse('$v') ?? 0;
  String _money(double v) => '\$${v.toStringAsFixed(2)}';

  Future<void> _load() async {
    setState(() { _loading = true; _error = null; });
    try {
      final responses = await Future.wait([
        ApiClient.get('/ventas?skip=0&limit=200'),
        ApiClient.get('/detalle-ventas?skip=0&limit=200'),
        ApiClient.get('/clientes?skip=0&limit=200'),
        ApiClient.get('/vendedores?skip=0&limit=200'),
        ApiClient.get('/puestos?skip=0&limit=200'),
      ]);
      for (final response in responses) { if (response is! List) throw Exception('Una ruta de ventas devolvió una respuesta no válida.'); }
      final ventas = (responses[0] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
      final detalles = (responses[1] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
      final clientes = (responses[2] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
      final vendedores = (responses[3] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
      final puestos = (responses[4] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
      final clienteMap = {for (final x in clientes) _int(x['id_cliente']): x};
      final vendedorMap = {for (final x in vendedores) _int(x['id_vendedor']): x};
      final puestoMap = {for (final x in puestos) _int(x['id_puesto']): x};
      final detallePorVenta = <int, List<Map<String, dynamic>>>{};
      for (final d in detalles) {
        final idVenta = _int(d['id_venta']);
        if (idVenta != null) (detallePorVenta[idVenta] ??= []).add(d);
      }
      for (final venta in ventas) {
        final id = _int(venta['id_venta']);
        final items = detallePorVenta[id] ?? [];
        final total = items.fold<double>(0, (sum, d) => sum + (_num(d['cantidad']) * _num(d['precio_unitario'])));
        final piezas = items.fold<int>(0, (sum, d) => sum + (_int(d['cantidad']) ?? 0));
        final cliente = clienteMap[_int(venta['id_cliente'])];
        final vendedor = vendedorMap[_int(venta['id_vendedor'])];
        final puesto = puestoMap[_int(venta['id_puesto'])];
        venta['_total_calculado'] = total;
        venta['_piezas'] = piezas;
        venta['_cliente_nombre'] = '${cliente?['nombre'] ?? ''} ${cliente?['apellido_p'] ?? ''}'.trim();
        venta['_vendedor_nombre'] = '${vendedor?['nombre'] ?? ''} ${vendedor?['apellido_paterno'] ?? ''}'.trim();
        venta['_puesto_nombre'] = puesto?['nombre_puesto'] ?? 'Puesto #${venta['id_puesto'] ?? '?'}';
      }
      ventas.sort((a, b) => (_int(b['id_venta']) ?? 0).compareTo(_int(a['id_venta']) ?? 0));
      if (!mounted) return;
      setState(() {
        _ventas = ventas;
        _totalGeneral = ventas.fold<double>(0, (sum, v) => sum + _num(v['_total_calculado']));
        _cantidadVentas = ventas.length;
        _totalPiezas = ventas.fold<int>(0, (sum, v) => sum + (_int(v['_piezas']) ?? 0));
        _loading = false;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() { _error = e.toString(); _loading = false; });
    }
  }

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: const Text('Ventas y reportes'), backgroundColor: Colors.brown.shade700, foregroundColor: Colors.white, actions: [IconButton(onPressed: _load, icon: const Icon(Icons.refresh), tooltip: 'Actualizar')]),
    body: _loading ? const Center(child: CircularProgressIndicator()) : _error != null
      ? Center(child: Padding(padding: const EdgeInsets.all(24), child: Column(mainAxisSize: MainAxisSize.min, children: [Text('No se pudieron cargar las ventas.\n$_error', textAlign: TextAlign.center), const SizedBox(height: 12), FilledButton(onPressed: _load, child: const Text('Reintentar'))])))
      : RefreshIndicator(onRefresh: _load, child: ListView(padding: const EdgeInsets.all(16), children: [
          Wrap(spacing: 10, runSpacing: 10, children: [
            _stat('Ventas registradas', '$_cantidadVentas', Icons.receipt_long),
            _stat('Importe calculado', _money(_totalGeneral), Icons.payments),
            _stat('Piezas vendidas', '$_totalPiezas', Icons.inventory_2),
          ]),
          const SizedBox(height: 18),
          const Text('Historial de ventas', style: TextStyle(fontSize: 19, fontWeight: FontWeight.bold)),
          const SizedBox(height: 8),
          if (_ventas.isEmpty) const Padding(padding: EdgeInsets.all(24), child: Text('Todavía no hay ventas registradas.', textAlign: TextAlign.center)),
          ..._ventas.map((v) => Card(child: ExpansionTile(
            leading: CircleAvatar(backgroundColor: Colors.brown.shade100, child: const Icon(Icons.point_of_sale, color: Colors.brown)),
            title: Text('Venta #${v['id_venta']} · ${_money(_num(v['_total_calculado']))}', style: const TextStyle(fontWeight: FontWeight.bold)),
            subtitle: Text('${v['fecha_hora'] ?? 'Fecha no disponible'}\n${v['_cliente_nombre'].toString().isEmpty ? 'Cliente sin nombre' : v['_cliente_nombre']}'),
            childrenPadding: const EdgeInsets.fromLTRB(16, 0, 16, 16),
            children: [
              _detail('Vendedor', v['_vendedor_nombre'].toString().isEmpty ? 'Sin nombre' : v['_vendedor_nombre'].toString()),
              _detail('Puesto', v['_puesto_nombre'].toString()),
              _detail('Método de pago', (v['metodo_pago'] ?? 'No especificado').toString()),
              _detail('Piezas', '${v['_piezas']}'),
              _detail('Total', _money(_num(v['_total_calculado']))),
            ],
          ))),
          const SizedBox(height: 12),
          const Text('El total se calcula sumando cantidad × precio_unitario de los detalles de cada venta; no se inventa un importe si no hay detalles.', style: TextStyle(color: Colors.grey, fontSize: 12)),
        ])),
  );

  Widget _stat(String title, String value, IconData icon) => SizedBox(width: 155, child: Card(color: Colors.brown.shade50, child: Padding(padding: const EdgeInsets.all(14), child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [Icon(icon, color: Colors.brown.shade700), const SizedBox(height: 10), Text(title, style: const TextStyle(fontSize: 12)), const SizedBox(height: 4), Text(value, style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 18))]))));
  Widget _detail(String label, String value) => Padding(padding: const EdgeInsets.only(top: 6), child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [SizedBox(width: 120, child: Text('$label:', style: const TextStyle(fontWeight: FontWeight.w600))), Expanded(child: Text(value))]));
}

class _AdminVendedoresScreenState

  bool _loading = true;
  String? _error;
  List<Map<String, dynamic>> _vendedores = [];

  int? _int(dynamic value) {
    if (value is int) return value;
    return int.tryParse(value?.toString() ?? '');
  }

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    if (mounted) {
      setState(() {
        _loading = true;
        _error = null;
      });
    }

    try {
      final respuesta = await ApiClient.get(
        '/admin/vendedores?skip=0&limit=200',
      );

      if (respuesta is! List) {
        throw Exception(
          'La API no devolvió una lista válida de vendedores.',
        );
      }

      final lista = respuesta
          .whereType<Map>()
          .map(
            (elemento) =>
                Map<String, dynamic>.from(elemento),
          )
          .toList();

      if (!mounted) return;

      setState(() {
        _vendedores = lista;
        _loading = false;
      });
    } catch (error) {
      if (!mounted) return;

      setState(() {
        _error = error.toString();
        _loading = false;
      });
    }
  }

  Future<void> _setActive(
    Map<String, dynamic> vendedor,
  ) async {
    final id = _int(vendedor['id_vendedor']);

    if (id == null) {
      _message(
        'La API no devolvió el ID del vendedor.',
        error: true,
      );
      return;
    }

    // "activo" representa el estado efectivo de la cuenta
    // devuelto por la ruta administrativa de FastAPI.
    final estaActivo = vendedor['activo'] == true;
    final siguienteEstado = !estaActivo;

    final confirmado = await showDialog<bool>(
      context: context,
      builder: (dialogContext) {
        return AlertDialog(
          title: Text(
            siguienteEstado
                ? 'Activar cuenta'
                : 'Desactivar cuenta',
          ),
          content: Text(
            siguienteEstado
                ? '¿Deseas activar la cuenta del vendedor #$id?'
                : '¿Deseas desactivar la cuenta del vendedor '
                    '#$id? El usuario no debería poder iniciar '
                    'sesión mientras su cuenta esté inactiva.',
          ),
          actions: [
            TextButton(
              onPressed: () =>
                  Navigator.pop(dialogContext, false),
              child: const Text('Cancelar'),
            ),
            FilledButton(
              onPressed: () =>
                  Navigator.pop(dialogContext, true),
              child: Text(
                siguienteEstado ? 'Activar' : 'Desactivar',
              ),
            ),
          ],
        );
      },
    );

    if (confirmado != true || !mounted) return;

    try {
      await ApiClient.patch(
        '/admin/vendedores/$id/estado',
        body: {'activo': siguienteEstado},
      );

      _message(
        siguienteEstado
            ? 'Cuenta activada correctamente.'
            : 'Cuenta desactivada correctamente.',
      );

      await _load();
    } catch (error) {
      _message(
        'No se pudo cambiar el estado: $error. '
        'Verifica que la ruta de FastAPI esté habilitada.',
        error: true,
      );
    }
  }

    Map<String, dynamic> vendedor,
  ) async {
    final id = _int(vendedor['id_vendedor']);

    if (id == null) {
      _message(
        'La API no devolvió el ID del vendedor.',
        error: true,
      );
      return;

      vendedor['nombre'],
      vendedor['apellido_paterno'],
      vendedor['apellido_materno'],
    ]
        .where(
          (parte) =>
              parte != null &&
              parte.toString().trim().isNotEmpty,
        )
        .join(' ');

    final rol =

      child: ListTile(
        leading: CircleAvatar(
          backgroundColor: activo
              ? Colors.green.shade100

          ),
        ),

          'Correo: $correo\n'
          'ID vendedor: ${id ?? '—'} · '
          '${activo ? 'Activo' : 'Inactivo'}\n'
          'Rol: $rol · Teléfono: $telefono',

        onTap: () => _openDetail(vendedor),

          mainAxisSize: MainAxisSize.min,

              tooltip: activo

              color: activo
                  ? Colors.red.shade700
                  : Colors.green.shade700,
            ),
            const Icon(Icons.chevron_right),
          ],
        ),
      ),

  final _contrasena = TextEditingController();
  final _telefono = TextEditingController();
  String _rol = 'Cajero';
  bool _saving = false;

  Future<void> _save() async {
    if (!_key.currentState!.validate()) return;
    setState(() => _saving = true);
    try {
      final result = await ApiClient.post('/registro/vendedor', body: {
        'nombre': _nombre.text.trim(),
        'apellido_paterno': _apellidoP.text.trim().isEmpty ? null : _apellidoP.text.trim(),

        'contrasena': _contrasena.text,
        'telefono': _telefono.text.trim().isEmpty ? null : _telefono.text.trim(),

      if (!mounted) return;

      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('No se pudo crear la cuenta: $e'), backgroundColor: Colors.red.shade700));

    }

      _field(_apellidoM, 'Apellido materno'),

      const SizedBox(height: 16),
      const Text('Datos de acceso', style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold)),
      const SizedBox(height: 12),
      _field(_correo, 'Correo electrónico', required: true, keyboard: TextInputType.emailAddress, validator: (v) => (v == null || !v.contains('@') || !v.contains('.')) ? 'Ingresa un correo válido' : null),
      _field(_contrasena, 'Contraseña temporal', required: true, obscure: true, validator: (v) => (v ?? '').length < 8 ? 'Usa al menos 8 caracteres' : null),
      DropdownButtonFormField<String>(value: _rol, decoration: const InputDecoration(labelText: 'Rol', border: OutlineInputBorder()),

      const SizedBox(height: 20),
      FilledButton.icon(onPressed: _saving ? null : _save,
        icon: _saving ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2)) : const Icon(Icons.person_add_alt_1),
        label: Text(_saving ? 'Creando cuenta...' : 'Crear cuenta de vendedor'), style: FilledButton.styleFrom(minimumSize: const Size.fromHeight(50))),
      const SizedBox(height: 12),
      const Text('La contraseña no se puede recuperar después. Entrega la contraseña temporal directamente al vendedor y pide que la cambie al tener esa función disponible.', style: TextStyle(color: Colors.grey)),
    ])),
  );

  Widget _field(TextEditingController controller, String label, {bool required = false, bool obscure = false, TextInputType? keyboard, String? Function(String?)? validator}) => Padding(
    padding: const EdgeInsets.only(bottom: 12), child: TextFormField(controller: controller, obscureText: obscure, keyboardType: keyboard,
      decoration: InputDecoration(labelText: label, border: const OutlineInputBorder()),
      validator: validator ?? (required ? (v) => (v == null || v.trim().isEmpty) ? 'Este campo es obligatorio' : null : null)),
  );
}

      if (result is! Map) throw Exception('La API devolvió un detalle con formato inesperado.');
      if (!mounted) return;
      setState(() { _data = Map<String, dynamic>.from(result); _loading = false; });
    } catch (e) {
      if (!mounted) return;
      setState(() { _error = e.toString(); _loading = false; });
    }
  }

  String _value(String key, [String fallback = 'No registrado']) {
    final value = _data?[key];
    if (value == null || value.toString().trim().isEmpty) return fallback;
    return value.toString();
  }

  Widget _section(String title, IconData icon, List<Widget> children) => Card(
    margin: const EdgeInsets.only(bottom: 12),
    child: Padding(padding: const EdgeInsets.all(16), child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [

      const Divider(height: 24),
      ...children,
    ])),
  );

  Widget _field(String label, String value, {IconData? icon}) => Padding(
    padding: const EdgeInsets.only(bottom: 12),
    child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
      if (icon != null) ...[Icon(icon, size: 19, color: Colors.grey.shade700), const SizedBox(width: 10)],
      SizedBox(width: icon == null ? 136 : 126, child: Text(label, style: TextStyle(color: Colors.grey.shade700))),
      Expanded(child: SelectableText(value.isEmpty ? 'No registrado' : value)),
    ]),
  );

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text('Vendedor #${widget.idVendedor}'), actions: [IconButton(onPressed: _load, icon: const Icon(Icons.refresh), tooltip: 'Actualizar')]),
    body: _loading
        ? const Center(child: CircularProgressIndicator())
        : _error != null
            ? Center(child: Padding(padding: const EdgeInsets.all(24), child: Column(mainAxisSize: MainAxisSize.min, children: [
                Text('No se pudo cargar la ficha del vendedor.\n$_error', textAlign: TextAlign.center),
                const SizedBox(height: 12),
                FilledButton(onPressed: _load, child: const Text('Reintentar')),
              ])))
            : RefreshIndicator(onRefresh: _load, child: ListView(padding: const EdgeInsets.all(16), children: [
                Center(child: CircleAvatar(radius: 38, backgroundColor: Colors.purple.shade100, child: Icon(Icons.badge_outlined, size: 42, color: Colors.purple.shade700))),
                const SizedBox(height: 12),
                Center(child: Text('${_value('nombre', '')} ${_value('apellido_paterno', '')} ${_value('apellido_materno', '')}'.trim(), textAlign: TextAlign.center, style: Theme.of(context).textTheme.headlineSmall?.copyWith(fontWeight: FontWeight.bold))),
                const SizedBox(height: 4),
                Center(child: Chip(label: Text(_data?['activo'] == true && _data?['usuario_activo'] != false ? 'Cuenta activa' : 'Cuenta inactiva'))),
                const SizedBox(height: 12),
                _section('Datos de acceso', Icons.key_outlined, [
                  _field('ID de usuario', _value('id_usuario'), icon: Icons.numbers),
                  _field('Correo', _value('correo'), icon: Icons.email_outlined),
                  _field('Acceso del usuario', _data?['usuario_activo'] == true ? 'Activo' : 'Inactivo', icon: Icons.verified_user_outlined),
                ]),
                _section('Información personal y laboral', Icons.person_outline, [
                  _field('ID de vendedor', _value('id_vendedor')),
                  _field('Nombre', _value('nombre')),
                  _field('Apellido paterno', _value('apellido_paterno')),
                  _field('Apellido materno', _value('apellido_materno')),
                  _field('Teléfono', _value('telefono'), icon: Icons.phone_outlined),
                  _field('Rol', _value('rol')),
                  _field('Estado del perfil', _data?['vendedor_activo'] == true ? 'Activo' : 'Inactivo'),
                ]),
                _section('Asignaciones de puesto', Icons.storefront_outlined, _asignaciones()),
              ])),
  );

  List<Widget> _asignaciones() {
    final raw = _data?['asignaciones'];
    if (raw is! List || raw.isEmpty) {
      return [const Text('No hay asignaciones de puesto registradas.')];
    }
    return raw.whereType<Map>().map((item) {
      final a = Map<String, dynamic>.from(item);
      final title = a['nombre_puesto']?.toString().trim().isNotEmpty == true
          ? a['nombre_puesto'].toString()
          : 'Puesto #${a['id_puesto'] ?? '—'}';
      final subtitle = [
        'Estado: ${a['estado'] ?? 'No especificado'}',
        'Turno: ${a['turno'] ?? 'No especificado'}',
        'Inicio: ${a['fecha_inicio'] ?? 'No registrada'}',
        'Fin: ${a['fecha_fin'] ?? 'Vigente'}',
      ].join('\n');
      return Padding(
        padding: const EdgeInsets.only(bottom: 8),
        child: ListTile(
          contentPadding: EdgeInsets.zero,
          leading: const Icon(Icons.store_outlined),
          title: Text(title),
          subtitle: Text(subtitle),
        ),
      );
    }).toList();
  }

/// Los pedidos de clientes son solicitudes; no son necesariamente ventas cobradas.
class AdminPedidosScreen extends StatefulWidget {
  const AdminPedidosScreen({super.key});
  @override
  State<AdminPedidosScreen> createState() => _AdminPedidosScreenState();
}

class _AdminPedidosScreenState extends State<AdminPedidosScreen> {
  late Future<List<Map<String, dynamic>>> _future = _fetch();

  Future<List<Map<String, dynamic>>> _fetch() async {
    final data = await ApiClient.get('/admin/pedidos-detallados?skip=0&limit=200');
    if (data is! List) throw Exception('La API no devolvió una lista de pedidos válida.');
    return data.whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
  }

  Future<void> _refresh() async {
    final next = _fetch();
    setState(() => _future = next);
    try { await next; } catch (_) {}
  }

  int _int(dynamic value) => value is num ? value.toInt() : int.tryParse('$value') ?? 0;
  double _num(dynamic value) => value is num ? value.toDouble() : double.tryParse('$value') ?? 0;
  String _text(dynamic value, [String fallback = 'No disponible']) {
    final text = value?.toString().trim() ?? '';
    return text.isEmpty ? fallback : text;
  }
  String _money(dynamic value) => '\$${_num(value).toStringAsFixed(2)}';
  bool _cancelado(Map<String, dynamic> row) => _text(row['estado'], '').toLowerCase().contains('cancel');
  bool _pendiente(Map<String, dynamic> row) => _text(row['estado'], '').toLowerCase().contains('pendiente');

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(
      title: const Text('Pedidos de clientes'),
      actions: [IconButton(onPressed: _refresh, tooltip: 'Actualizar', icon: const Icon(Icons.refresh))],
    ),
    body: FutureBuilder<List<Map<String, dynamic>>>(
      future: _future,
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) return const Center(child: CircularProgressIndicator());
        if (snapshot.hasError) {
          return Center(child: Padding(padding: const EdgeInsets.all(24), child: Column(mainAxisSize: MainAxisSize.min, children: [
            const Icon(Icons.cloud_off_outlined, size: 44),
            const SizedBox(height: 10),
            Text('No se pudieron consultar los pedidos.\n${snapshot.error}', textAlign: TextAlign.center),
            const SizedBox(height: 12),
            FilledButton.icon(onPressed: _refresh, icon: const Icon(Icons.refresh), label: const Text('Reintentar')),
          ])));
        }
        final orders = snapshot.data ?? <Map<String, dynamic>>[];
        final pending = orders.where(_pendiente).length;
        final cancelled = orders.where(_cancelado).length;
        final total = orders.fold<double>(0, (sum, item) => sum + _num(item['total']));
        return RefreshIndicator(
          onRefresh: _refresh,
          child: ListView(physics: const AlwaysScrollableScrollPhysics(), padding: const EdgeInsets.all(16), children: [
            const Text('Solicitudes realizadas desde la app', style: TextStyle(fontSize: 13, color: Colors.grey)),
            const SizedBox(height: 10),
            Wrap(spacing: 8, runSpacing: 8, children: [
              _stat('Pedidos', '${orders.length}', Icons.receipt_long),
              _stat('Pendientes', '$pending', Icons.hourglass_top),
              _stat('Cancelados', '$cancelled', Icons.cancel_outlined),
              _stat('Importe solicitado', _money(total), Icons.payments_outlined),
            ]),
            const SizedBox(height: 18),
            if (orders.isEmpty) const Padding(padding: EdgeInsets.all(24), child: Column(children: [
              Icon(Icons.inbox_outlined, size: 48, color: Colors.grey),
              SizedBox(height: 8),
              Text('Todavía no hay pedidos registrados.', textAlign: TextAlign.center),
              SizedBox(height: 4),
              Text('Cuando un cliente compre desde el catálogo, aparecerá aquí.', textAlign: TextAlign.center, style: TextStyle(color: Colors.grey)),
            ])),
            ...orders.map((order) {
              final items = order['items'] is List ? (order['items'] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList() : <Map<String, dynamic>>[];
              final status = _text(order['estado'], 'Sin estado');
              final statusColor = _cancelado(order) ? Colors.red : _pendiente(order) ? Colors.orange.shade800 : Colors.green.shade700;
              return Card(margin: const EdgeInsets.only(bottom: 12), child: ExpansionTile(
                leading: CircleAvatar(backgroundColor: statusColor.withValues(alpha: .12), child: Icon(Icons.shopping_bag_outlined, color: statusColor)),
                title: Text('Pedido #${_int(order['id_pedido'])} · ${_money(order['total'])}', style: const TextStyle(fontWeight: FontWeight.bold)),
                subtitle: Text('${_text(order['fecha_solicitud'], 'Sin fecha')}\n${_text(order['cliente'], 'Cliente sin nombre')}'),
                trailing: Chip(label: Text(status), visualDensity: VisualDensity.compact),
                childrenPadding: const EdgeInsets.fromLTRB(16, 0, 16, 16),
                children: [
                  _detail('Correo cliente', _text(order['cliente_correo'])),
                  _detail('Teléfono cliente', _text(order['cliente_telefono'])),
                  _detail('Artesano', _text(order['artesano'])),
                  _detail('Teléfono artesano', _text(order['artesano_telefono'])),
                  _detail('Puesto', '${_text(order['puesto'])} · ${_text(order['puesto_numero'], 'sin número')}'),
                  const Divider(height: 20),
                  const Align(alignment: Alignment.centerLeft, child: Text('Productos del pedido', style: TextStyle(fontWeight: FontWeight.bold))),
                  if (items.isEmpty) const ListTile(dense: true, title: Text('Este pedido no tiene partidas registradas.')),
                  ...items.map((item) => ListTile(
                    dense: true,
                    contentPadding: EdgeInsets.zero,
                    title: Text('${_int(item['cantidad'])} × ${_text(item['producto'], 'Producto')}'),
                    subtitle: Text('${_money(item['precio_unitario'])} por unidad'),
                    trailing: Text(_money(item['subtotal']), style: const TextStyle(fontWeight: FontWeight.w600)),
                  )),
                  _detail('Anticipo', _money(order['anticipo'])),
                  if (_text(order['observaciones'], '').isNotEmpty) _detail('Observaciones / pago', _text(order['observaciones'], '')),
                  const SizedBox(height: 4),
                  const Text('Un pedido se muestra aquí; solo aparece en Ventas y Reportes cuando se registra una venta en caja.', style: TextStyle(color: Colors.grey, fontSize: 12)),
                ],
              ));
            }),
          ]),
        );
      },
    ),
  );

  Widget _stat(String title, String value, IconData icon) => SizedBox(width: 155, child: Card(child: Padding(padding: const EdgeInsets.all(12), child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [Icon(icon), const SizedBox(height: 6), Text(value, style: const TextStyle(fontSize: 18, fontWeight: FontWeight.bold)), Text(title, style: const TextStyle(fontSize: 12))]))));
  Widget _detail(String label, String value) => Padding(padding: const EdgeInsets.only(top: 6), child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [SizedBox(width: 130, child: Text('$label:', style: const TextStyle(fontWeight: FontWeight.w600))), Expanded(child: Text(value))]));
}

  const _CatalogField({

    this.numeric = false,
    this.maxLines = 1,
    this.provider = false,
  });
}

class _CatalogSpec {
  final String endpoint;
  final String idKey;
  final String title;
  final List<_CatalogField> fields;

  const _CatalogSpec({
    required this.endpoint,
    required this.idKey,
    required this.title,
    required this.fields,
  });
}

class AdminCatalogosScreen extends StatefulWidget {
  const AdminCatalogosScreen({super.key});

  @override
  State<AdminCatalogosScreen> createState() => _AdminCatalogosScreenState();
}

class _AdminCatalogosScreenState extends State<AdminCatalogosScreen> {
  static const List<_CatalogSpec> _specs = [
    _CatalogSpec(
      endpoint: 'categorias-estilo',
      idKey: 'id_estilo',
      title: 'Estilos',
      fields: [
        _CatalogField(key: 'nombre_estilo', label: 'Nombre del estilo', required: true),
        _CatalogField(key: 'descripcion', label: 'Descripción', maxLines: 3),
      ],
    ),
    _CatalogSpec(
      endpoint: 'calidades',
      idKey: 'id_calidad',
      title: 'Calidades',
      fields: [
        _CatalogField(key: 'nombre_calidad', label: 'Nombre de calidad', required: true),
        _CatalogField(key: 'dias_elaboracion_estimados', label: 'Días estimados de elaboración', numeric: true),
        _CatalogField(key: 'numero_partidas', label: 'Número de partidas', numeric: true),
      ],
    ),
    _CatalogSpec(
      endpoint: 'materiales',
      idKey: 'id_material',
      title: 'Materiales',
      fields: [
        _CatalogField(key: 'nombre_material', label: 'Nombre del material', required: true),
        _CatalogField(key: 'unidad_medida', label: 'Unidad de medida'),
        _CatalogField(key: 'id_proveedor', label: 'Proveedor', required: true, numeric: true, provider: true),
      ],
    ),
  ];

  bool _loading = true;
  String? _error;
  final Map<String, List<Map<String, dynamic>>> _records = {};
  List<Map<String, dynamic>> _providers = [];

  int? _toInt(dynamic value) => value is int ? value : int.tryParse('$value');

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<List<Map<String, dynamic>>> _readList(String endpoint) async {
    final data = await ApiClient.get('/$endpoint?skip=0&limit=200');
    if (data is! List) throw Exception('Respuesta inválida de /$endpoint.');
    return data
        .whereType<Map>()
        .map((item) => Map<String, dynamic>.from(item))
        .toList();
  }

  Future<void> _load() async {
    if (mounted) setState(() { _loading = true; _error = null; });
    try {
      final results = await Future.wait([
        for (final spec in _specs) _readList(spec.endpoint),
        _readList('proveedores'),
      ]);
      if (!mounted) return;
      setState(() {
        for (var i = 0; i < _specs.length; i++) {
          _records[_specs[i].endpoint] = results[i];
        }
        _providers = results.last;
        _loading = false;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() { _error = e.toString(); _loading = false; });
    }
  }

  void _message(String value, {bool error = false}) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(
      content: Text(value),
      backgroundColor: error ? Colors.red.shade700 : null,
    ));
  }

  String _displayName(_CatalogSpec spec, Map<String, dynamic> item) {
    final fields = spec.fields.where((f) => !f.provider).toList();
    if (fields.isEmpty) return '${spec.title} #${item[spec.idKey] ?? '—'}';
    final value = item[fields.first.key]?.toString().trim() ?? '';
    return value.isEmpty ? '${spec.title} #${item[spec.idKey] ?? '—'}' : value;
  }

  String _summary(_CatalogSpec spec, Map<String, dynamic> item) {
    final parts = <String>[];
    for (final field in spec.fields) {
      if (field.provider) {
        final providerId = _toInt(item[field.key]);
        final supplier = _providers.where((p) => _toInt(p['id_proveedor']) == providerId).toList();
        final providerName = supplier.isEmpty
            ? 'Proveedor #${providerId ?? '—'}'
            : (supplier.first['nombre_proveedor'] ?? supplier.first['nombre'] ?? 'Proveedor #$providerId').toString();
        parts.add('${field.label}: $providerName');
      } else {
        final value = item[field.key];
        if (value != null && value.toString().trim().isNotEmpty) {
          parts.add('${field.label}: $value');
        }
      }
    }
    return parts.join(' · ');
  }

  Future<void> _form(_CatalogSpec spec, [Map<String, dynamic>? existing]) async {
    final controllers = <String, TextEditingController>{};
    for (final field in spec.fields.where((f) => !f.provider)) {
      controllers[field.key] = TextEditingController(
        text: existing?[field.key]?.toString() ?? '',
      );
    }
    int? providerId = _toInt(existing?['id_proveedor']);
    final providerIds = _providers.map((p) => _toInt(p['id_proveedor'])).whereType<int>().toSet();
    if (!providerIds.contains(providerId)) {
      if (existing == null) {
        providerId = _providers.isEmpty ? null : _toInt(_providers.first['id_proveedor']);
      } else if (_providers.isNotEmpty) {
        providerId = _toInt(_providers.first['id_proveedor']);
      }
    }

    if (spec.fields.any((f) => f.provider) && _providers.isEmpty && existing == null) {
      _dispose(controllers);
      _message('Primero registra al menos un proveedor para poder crear materiales.', error: true);
      return;
    }

    final ok = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => StatefulBuilder(
        builder: (dialogContext, dialogSetState) => AlertDialog(
          title: Text(existing == null ? 'Nuevo registro · ${spec.title}' : 'Editar · ${spec.title}'),
          content: SizedBox(
            width: 460,
            child: SingleChildScrollView(
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  for (final field in spec.fields)
                    if (field.provider)
                      DropdownButtonFormField<int>(
                        value: providerId,
                        isExpanded: true,
                        decoration: InputDecoration(labelText: '${field.label}${field.required ? ' *' : ''}', border: const OutlineInputBorder()),
                        items: [
                          if (providerId != null && !_providers.any((p) => _toInt(p['id_proveedor']) == providerId))
                            DropdownMenuItem<int>(value: providerId, child: Text('Proveedor #$providerId (existente)')),
                          ..._providers.map((p) {
                            final id = _toInt(p['id_proveedor']);
                            if (id == null) return null;
                            final name = p['nombre_proveedor'] ?? p['nombre'] ?? 'Proveedor #$id';
                            return DropdownMenuItem<int>(value: id, child: Text('$name'));
                          }).whereType<DropdownMenuItem<int>>(),
                        ],
                        onChanged: (value) => dialogSetState(() => providerId = value),
                      )
                    else
                      Padding(
                        padding: const EdgeInsets.only(bottom: 12),
                        child: TextField(
                          controller: controllers[field.key],
                          keyboardType: field.numeric ? TextInputType.number : TextInputType.text,
                          maxLines: field.maxLines,
                          decoration: InputDecoration(
                            labelText: '${field.label}${field.required ? ' *' : ''}',
                            border: const OutlineInputBorder(),
                          ),
                        ),
                      ),
                ],
              ),
            ),
          ),
          actions: [
            TextButton(onPressed: () => Navigator.pop(dialogContext, false), child: const Text('Cancelar')),
            FilledButton(onPressed: () => Navigator.pop(dialogContext, true), child: const Text('Guardar')),
          ],
        ),
      ),
    );

    if (ok != true) {
      _dispose(controllers);
      return;
    }

    final body = <String, dynamic>{};
    String? missing;
    for (final field in spec.fields) {
      if (field.provider) {
        if (providerId == null) {
          missing = field.label;
        } else {
          body[field.key] = providerId;
        }
        continue;
      }
      final text = controllers[field.key]!.text.trim();
      if (field.required && text.isEmpty) {
        missing = field.label;
        break;
      }
      if (field.numeric) {
        if (text.isEmpty) {
          body[field.key] = null;
        } else {
          final parsed = int.tryParse(text);
          if (parsed == null || parsed < 0) {
            missing = 'un valor numérico válido para ${field.label}';
            break;
          }
          body[field.key] = parsed;
        }
      } else {
        body[field.key] = text.isEmpty ? null : text;
      }
    }
    _dispose(controllers);

    if (missing != null) {
      _message('Revisa el campo: $missing.', error: true);
      return;
    }

    try {
      final id = _toInt(existing?[spec.idKey]);
      if (id == null) {
        await ApiClient.post('/${spec.endpoint}', body: body);
        _message('Registro creado. El ID lo asignó la base de datos.');
      } else {
        await ApiClient.patch('/${spec.endpoint}/$id', body: body);
        _message('Registro actualizado.');
      }
      await _load();
    } catch (e) {
      _message('No se pudo guardar ${spec.title.toLowerCase()}: $e', error: true);
    }
  }

  void _dispose(Map<String, TextEditingController> controllers) {
    for (final controller in controllers.values) {
      controller.dispose();
    }
  }

  Widget _buildTab(_CatalogSpec spec) {
    final items = _records[spec.endpoint] ?? [];
    return RefreshIndicator(
      onRefresh: _load,
      child: ListView(
        physics: const AlwaysScrollableScrollPhysics(),
        padding: const EdgeInsets.all(12),
        children: [
          Card(
            child: Padding(
              padding: const EdgeInsets.all(12),
              child: Text('${spec.title}: ${items.length} registros'),
            ),
          ),
          if (items.isEmpty)
            const Padding(
              padding: EdgeInsets.all(24),
              child: Text('No hay registros en este catálogo. Agrega el primero.', textAlign: TextAlign.center),
            ),
          for (final item in items)
            Card(
              child: ListTile(
                title: Text(_displayName(spec, item)),
                subtitle: Text('ID ${item[spec.idKey] ?? '—'} · ${_summary(spec, item)}'),
                isThreeLine: true,
                trailing: IconButton(
                  tooltip: 'Editar registro',
                  icon: const Icon(Icons.edit_outlined),
                  onPressed: () => _form(spec, item),
                ),
              ),
            ),
          const SizedBox(height: 72),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) => DefaultTabController(
        length: _specs.length,
        child: Scaffold(
          appBar: AppBar(
            title: const Text('Catálogos del sistema'),
            backgroundColor: Colors.brown.shade700,
            foregroundColor: Colors.white,
            actions: [IconButton(onPressed: _load, icon: const Icon(Icons.refresh), tooltip: 'Actualizar')],
            bottom: const TabBar(
              isScrollable: true,
              tabs: [Tab(text: 'Estilos'), Tab(text: 'Calidades'), Tab(text: 'Materiales')],
            ),
          ),
          body: _loading
              ? const Center(child: CircularProgressIndicator())
              : _error != null
                  ? Center(
                      child: Padding(
                        padding: const EdgeInsets.all(24),
                        child: Column(
                          mainAxisSize: MainAxisSize.min,
                          children: [
                            Text('No se pudieron cargar los catálogos.\n$_error', textAlign: TextAlign.center),
                            const SizedBox(height: 12),
                            FilledButton(onPressed: _load, child: const Text('Reintentar')),
                          ],
                        ),
                      ),
                    )
                  : TabBarView(children: _specs.map(_buildTab).toList()),
          floatingActionButton: Builder(
            builder: (context) => FloatingActionButton.extended(
              onPressed: () {
                final index = DefaultTabController.of(context).index;
                _form(_specs[index]);
              },
              icon: const Icon(Icons.add),
              label: const Text('Agregar'),
            ),
          ),
        ),
      );
}

import 'package:flutter/material.dart';

class AdminArtesanoFormScreen extends StatefulWidget {
  const AdminArtesanoFormScreen({super.key});

  @override
  State<AdminArtesanoFormScreen> createState() => _AdminArtesanoFormScreenState();
}

class _AdminArtesanoFormScreenState extends State<AdminArtesanoFormScreen> {
  final _formKey = GlobalKey<FormState>();
  final _nombre = TextEditingController();
  final _apellidoP = TextEditingController();
  final _apellidoM = TextEditingController();
  final _correo = TextEditingController();
  final _contrasena = TextEditingController();
  final _telefono = TextEditingController();
  final _biografia = TextEditingController();
  final _experiencia = TextEditingController(text: '0');
  List<Map<String, dynamic>> _cuevas = [];
  List<Map<String, dynamic>> _puestos = [];
  int? _idCueva;
  int? _idPuesto;
  bool _loadingOptions = true;
  bool _saving = false;
  String? _loadError;

  @override
  void initState() {
    super.initState();
    _loadOptions();
  }

  Future<void> _loadOptions() async {
    try {
      final results = await Future.wait([
        ApiClient.get('/cuevas?skip=0&limit=200'),
        ApiClient.get('/puestos?skip=0&limit=200'),
      ]);
      if (results[0] is! List || results[1] is! List) {
        throw Exception('La API no devolvió los catálogos de cuevas y puestos.');
      }
      if (!mounted) return;
      setState(() {
        _cuevas = (results[0] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
        _puestos = (results[1] as List).whereType<Map>().map((e) => Map<String, dynamic>.from(e)).toList();
        _idCueva = _cuevas.isNotEmpty ? _toInt(_cuevas.first['id_cueva']) : null;
        _idPuesto = _puestos.isNotEmpty ? _toInt(_puestos.first['id_puesto']) : null;
        _loadingOptions = false;
      });
    } catch (e) {
      if (!mounted) return;
      setState(() { _loadingOptions = false; _loadError = e.toString(); });
    }
  }

  int? _toInt(dynamic value) => value is int ? value : int.tryParse('$value');

  Future<void> _save() async {
    if (!_formKey.currentState!.validate()) return;
    if (_idCueva == null || _idPuesto == null) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Primero registra al menos una cueva y un puesto.')));
      return;
    }
    setState(() => _saving = true);
    try {
      final experience = int.tryParse(_experiencia.text.trim()) ?? 0;
      final result = await ApiClient.post('/registro/artesano', body: {
        'nombre': _nombre.text.trim(),
        'apellido_paterno': _apellidoP.text.trim().isEmpty ? null : _apellidoP.text.trim(),
        'apellido_materno': _apellidoM.text.trim().isEmpty ? null : _apellidoM.text.trim(),
        'correo': _correo.text.trim(),
        'contrasena': _contrasena.text,
        'telefono': _telefono.text.trim().isEmpty ? null : _telefono.text.trim(),
        'biografia': _biografia.text.trim().isEmpty ? null : _biografia.text.trim(),
        'anios_experiencia': experience,
        'id_cueva': _idCueva,
        'id_puesto': _idPuesto,
      });
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Cuenta de artesano creada. ID: ${result is Map ? result['id_artesano'] ?? 'OK' : 'OK'}')));
      Navigator.pop(context, true);
    } catch (e) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('No se pudo crear la cuenta: $e'), backgroundColor: Colors.red));
    } finally {
      if (mounted) setState(() => _saving = false);
    }
  }

  @override
  void dispose() {
    _nombre.dispose(); _apellidoP.dispose(); _apellidoM.dispose(); _correo.dispose();
    _contrasena.dispose(); _telefono.dispose(); _biografia.dispose(); _experiencia.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Crear cuenta de artesano'), backgroundColor: Colors.brown.shade700, foregroundColor: Colors.white),
      body: _loadingOptions
          ? const Center(child: CircularProgressIndicator())
          : _loadError != null
              ? Center(child: Padding(padding: const EdgeInsets.all(24), child: Column(mainAxisSize: MainAxisSize.min, children: [Text('No se pudieron cargar cuevas y puestos.\n$_loadError', textAlign: TextAlign.center), const SizedBox(height: 12), FilledButton(onPressed: () { setState(() { _loadingOptions = true; _loadError = null; }); _loadOptions(); }, child: const Text('Reintentar'))])))
              : (_cuevas.isEmpty || _puestos.isEmpty)
                  ? const Center(child: Padding(padding: EdgeInsets.all(24), child: Text('Para crear una cuenta de artesano necesitas registrar primero al menos una cueva y un puesto en el panel de administración.', textAlign: TextAlign.center)))
                  : Form(

class ItemPedido {

    required this.cantidad,
    required this.precioUnitario,
    required this.subtotal,

  factory ItemPedido.fromJson(Map<String, dynamic> json) {

/// Cliente HTTP común para URL, timeout, JSON y subida multipart.
class ApiClient {
  ApiClient._();
  static final http.Client _http = http.Client();
  static const Duration _timeout = Duration(seconds: 15);
  static const Duration _uploadTimeout = Duration(seconds: 45);

  static Uri _uri(String path, [Map<String, String>? query]) {
    final normalized = path.startsWith('/') ? path : '/$path';
    return Uri.parse('${ApiConfig.baseUrl}$normalized').replace(queryParameters: query);
  }

  static Map<String, String> get _jsonHeaders => const {
    'Accept': 'application/json',
    'Content-Type': 'application/json; charset=utf-8',
  };

  static Future<dynamic> get(String path, {Map<String, String>? query}) =>
      _send(() => _http.get(_uri(path, query), headers: _jsonHeaders));

  static Future<dynamic> post(String path, {Object? body}) =>
      _send(() => _http.post(_uri(path), headers: _jsonHeaders, body: jsonEncode(body ?? <String, dynamic>{})));

  static Future<dynamic> put(String path, {Object? body}) =>
      _send(() => _http.put(_uri(path), headers: _jsonHeaders, body: jsonEncode(body ?? <String, dynamic>{})));

  static Future<dynamic> patch(String path, {Object? body}) =>
      _send(() => _http.patch(_uri(path), headers: _jsonHeaders, body: jsonEncode(body ?? <String, dynamic>{})));

  static Future<dynamic> delete(String path) =>
      _send(() => _http.delete(_uri(path), headers: _jsonHeaders));

  /// Envía bytes de una imagen como multipart/form-data, sin Base64 en JSON.
  static Future<dynamic> uploadFile(
    String path, {
    required Uint8List bytes,
    required String filename,
    String fieldName = 'archivo',
  }) async {
    late http.Response response;
    try {
      final request = http.MultipartRequest('POST', _uri(path));
      request.headers['Accept'] = 'application/json';
      request.files.add(http.MultipartFile.fromBytes(fieldName, bytes, filename: filename));
      final streamed = await request.send().timeout(_uploadTimeout);
      response = await http.Response.fromStream(streamed).timeout(_timeout);
    } on TimeoutException {
      throw const ApiException('La subida tardó demasiado. Intenta con una imagen más pequeña.');
    } on Exception catch (e) {
      if (e is ApiException) rethrow;
      throw ApiException('No se pudo subir el archivo. Verifica la conexión con la API.');
    }
    return _parseResponse(response);
  }

  static Future<dynamic> _send(Future<http.Response> Function() request) async {
    late http.Response response;
    try {
      response = await request().timeout(_timeout);
    } on TimeoutException {
      throw const ApiException('El servidor tardó demasiado en responder.');
    } on Exception catch (e) {
      if (e is ApiException) rethrow;
      throw ApiException('No se pudo conectar con la API. Verifica la URL, que FastAPI esté activo y la red.');
    }
    return _parseResponse(response);
  }

  static dynamic _parseResponse(http.Response response) {
    dynamic data;
    if (response.body.trim().isNotEmpty) {
      try { data = jsonDecode(utf8.decode(response.bodyBytes)); }
      catch (_) { data = response.body; }
    }
    if (response.statusCode < 200 || response.statusCode >= 300) {
      final detail = data is Map ? (data['detail'] ?? data['message']) : null;
      final message = detail is String ? detail : 'Error HTTP ${response.statusCode}';
      throw ApiException(message, statusCode: response.statusCode, details: data);
    }
    return data;
  }
}

import 'package:flutter/foundation.dart' show defaultTargetPlatform, kIsWeb, TargetPlatform;

/// URL central de FastAPI. Sobrescribir con --dart-define=API_BASE_URL=...
class ApiConfig {
  static const String _override = String.fromEnvironment('API_BASE_URL');

  static String get baseUrl {
    if (_override.trim().isNotEmpty) {
      return _override.trim().replaceAll(RegExp(r'/+$'), '');
    }
    if (kIsWeb) return 'http://localhost:8000';
    if (defaultTargetPlatform == TargetPlatform.android) {
      return 'http://192.168.1.67:8000'; // Emulador Android Studio
    }
    return 'http://localhost:8000'; // Escritorio/iOS Simulator
  }
}
